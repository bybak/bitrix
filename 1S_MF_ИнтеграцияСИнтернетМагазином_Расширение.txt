&Вместо("ЗаписатьДокумент")
Процедура MF_ВместоЗаписатьДокумент(Параметры, ДокументОбъект, ЗаписыватьВсегда = Ложь)

	ПропуститьЗапись = Ложь;
	НомерСайта = "";

	Если Не ЗаписыватьВсегда Тогда
		НомерСайта = MF_NumbersOrdersSaitServer.MF_НомерСайтаПоДокументуОбмена(ДокументОбъект);
		СуществующийЗаказ = Неопределено;
		Если НомерСайта <> "" Тогда
			СуществующийЗаказ = MF_NumbersOrdersSaitServer.MF_НайтиЛучшийЗаказПоНомеруСайта(НомерСайта);
		КонецЕсли;

		Если (Не ДокументОбъект.Ссылка.Пустая() И НЕ ОбновлятьЗагруженныеДокументы(Параметры, ДокументОбъект.Проведен)) Тогда
			ПропуститьЗапись = Истина;
		ИначеЕсли ЗначениеЗаполнено(СуществующийЗаказ)
			И (ДокументОбъект.Ссылка.Пустая() Или ДокументОбъект.Ссылка = СуществующийЗаказ)
			И MF_NumbersOrdersSaitServer.MF_ВзятьСвойстваОтменыИзКэша(НомерСайта) <> Неопределено Тогда
			ПропуститьЗапись = Истина;
		КонецЕсли;
	КонецЕсли;

	Если ПропуститьЗапись Тогда
		Если НомерСайта = "" Тогда
			НомерСайта = MF_NumbersOrdersSaitServer.MF_НомерСайтаПоДокументуОбмена(ДокументОбъект);
		КонецЕсли;
		MF_NumbersOrdersSaitServer.АвтоPaykeeperЛог(ДокументОбъект.Ссылка, "хук3 v18.6: пропуск записи " + НомерСайта);
		MF_NumbersOrdersSaitServer.MF_ПрименитьОтменуПриПропускеЗаписиОбмена(ДокументОбъект);
		Возврат;
	КонецЕсли;

	MF_NumbersOrdersSaitServer.MF_ЗаписатьДокументОбменаВместо(ДокументОбъект);

	Если НЕ ЗначениеЗаполнено(ДокументОбъект.Дата) Тогда
		ДокументОбъект.Дата = ТекущаяДатаСеанса();
	КонецЕсли;

&Вместо("ЗаписатьЗаказыПокупателейССайта")
Процедура MF_ВместоЗаписатьЗаказыПокупателейССайта(ДокументXDTO, ДокументОбъект, Параметры)

	ПродолжитьВызов(ДокументXDTO, ДокументОбъект, Параметры);

	НомерСайта = "";
	Попытка
		Если ДокументОбъект.ДополнительныеСвойства.Свойство("MF_NомерСайта") Тогда
			НомерСайта = СокрЛП(Строка(ДокументОбъект.ДополнительныеСвойства.MF_NомерСайта));
		КонецЕсли;
	Исключение
	КонецПопытки;
	Если НомерСайта = "" Тогда
		Попытка
			НомерСайта = СокрЛП(Строка(ДокументXDTO.Номер));
		Исключение
		КонецПопытки;
	КонецЕсли;
	Если НомерСайта <> "" Тогда
		MF_NumbersOrdersSaitServer.MF_ЗавершитьЗаказПослеОбмена(ДокументXDTO, ДокументОбъект);
	КонецЕсли;

КонецПроцедуры

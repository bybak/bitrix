// /create -> POST -> handler Orders

//
// ВАЖНО: этот вариант заполняет обязательные для проведения поля:
// - Валюта, Курс, Кратность
// - Операция
// - Налогообложение
// - Дата поступления
// - % НДС в строках табличной части
// ===============================

&НаСервере
Функция Orders(Запрос) Экспорт

	// ---------------- AUTH ----------------
	ОжидаемыйТокен = "wM85Jgw0kkdXRqp8EHQMwGsMYsvOPAvVnYM_nIVr6qnZ3EcgUI87SLf-mN96wAl5";
	Токен = HeaderValue(Запрос, "X-MF-Token");

	Если НЕ ПустаяСтрока(ОжидаемыйТокен) Тогда
		Если Токен <> ОжидаемыйТокен Тогда
			Возврат ОтветОшибка(401, "unauthorized");
		КонецЕсли;
	КонецЕсли;

	// ---------------- DEBUG (metadata dump) ----------------
	DebugMode = DebugModeFromRequest(Запрос);
	Если DebugMode = "META" Тогда
		Возврат DebugDescribeOrderDoc();
	ИначеЕсли DebugMode = "VALUES" Тогда
		Возврат DebugDescribeLookups();
	КонецЕсли;

	// ---------------- BODY ----------------
	Тело = ПолучитьТелоКакСтроку(Запрос);
	Если ПустаяСтрока(Тело) Тогда
		Возврат ОтветОшибка(400, "empty_body");
	КонецЕсли;

	Данные = JSONИзСтроки(Тело);
	Если Данные = Неопределено Тогда
		Возврат ОтветОшибка(400, "bad_json");
	КонецЕсли;

	ЗаказДанные = JsonGet(Данные, "order", Неопределено);
	Если ЗаказДанные = Неопределено Тогда
		Возврат ОтветОшибка(400, "no_order");
	КонецЕсли;

	ВнешнийИДЗаказа = СокрЛП(Строка(JsonGet(ЗаказДанные, "external_id", "")));
	Если ПустаяСтрока(ВнешнийИДЗаказа) Тогда
		Возврат ОтветОшибка(400, "no_external_id");
	КонецЕсли;

	// person_type из Bitrix
	PersonTypeId = ToNumber(JsonGet(ЗаказДанные, "person_type_id", 0));
	PersonTypeName = СокрЛП(Строка(JsonGet(ЗаказДанные, "person_type_name", "")));

	// customer
	Клиент = JsonGet(ЗаказДанные, "customer", Неопределено);
	Email = СокрЛП(Строка(JsonGet(Клиент, "email", "")));
	Телефон = СокрЛП(Строка(JsonGet(Клиент, "phone", "")));
	Имя = СокрЛП(Строка(JsonGet(Клиент, "name", "")));
	Адрес = СокрЛП(Строка(JsonGet(Клиент, "address", "")));

	// ---------------- REQUIRED FIELDS (UNF) ----------------
	ВалютаКод = СокрЛП(Строка(JsonGet(ЗаказДанные, "currency", "")));
	Если ПустаяСтрока(ВалютаКод) Тогда ВалютаКод = "RUB"; КонецЕсли;

	Курс = ToNumber(JsonGet(ЗаказДанные, "currency_rate", JsonGet(ЗаказДанные, "rate", 1)));
	Если Курс <= 0 Тогда Курс = 1; КонецЕсли;

	Кратность = ToNumber(JsonGet(ЗаказДанные, "currency_multiplicity", JsonGet(ЗаказДанные, "multiplicity", 1)));
	Если Кратность <= 0 Тогда Кратность = 1; КонецЕсли;

	ОперацияСтрока = СокрЛП(Строка(JsonGet(ЗаказДанные, "operation", "")));
	Если ПустаяСтрока(ОперацияСтрока) Тогда ОперацияСтрока = "Продажа"; КонецЕсли;

	НалогообложениеСтрока = СокрЛП(Строка(JsonGet(ЗаказДанные, "taxation", "")));
	Если ПустаяСтрока(НалогообложениеСтрока) Тогда НалогообложениеСтрока = "ОСН"; КонецЕсли;

	ДатаПоступленияСтрока = СокрЛП(Строка(JsonGet(ЗаказДанные, "receipt_date_local", "")));

	// ---------------- IDEMPOTENCY ----------------
	СсылкаСуществ = НайтиЗаказПоВнешнемуИД(ВнешнийИДЗаказа);

	Если СсылкаСуществ <> Неопределено Тогда

		Попытка
			ОбъектСуществ = СсылкаСуществ.ПолучитьОбъект();
			// Если документ ранее был помечен на удаление — снимаем пометку при повторной синхронизации,
			// иначе он "не появляется" в списках и выглядит как не созданный.
			Попытка
				ОбъектСуществ.ПометкаУдаления = Ложь;
			Исключение
			КонецПопытки;
			BackfillVisibilityFields(ОбъектСуществ, PersonTypeId, PersonTypeName, Имя, Телефон, Email);
			BackfillShipmentFields(ОбъектСуществ, ОбъектСуществ.Дата);
			ApplyUnfRequiredFields(ОбъектСуществ, ВалютаКод, Курс, Кратность, ОперацияСтрока, НалогообложениеСтрока, ДатаПоступленияСтрока);
			ApplyVatFromJsonToLines(ОбъектСуществ, ЗаказДанные);
			ОбъектСуществ.Записать();
		Исключение
		КонецПопытки;

		НомерСуществ = "";
		ДатаСуществ = ТекущаяДата();
		ПометкаУдаления = Ложь;
		НавСсылка = "";

		Попытка
			ОбъектСуществ2 = СсылкаСуществ.ПолучитьОбъект();
			НомерСуществ = ОбъектСуществ2.Номер;
			ДатаСуществ = ОбъектСуществ2.Дата;
			ПометкаУдаления = ОбъектСуществ2.ПометкаУдаления;
		Исключение
		КонецПопытки;

		Попытка
			НавСсылка = ПолучитьНавигационнуюСсылку(СсылкаСуществ);
		Исключение
		КонецПопытки;

		ТелоОтвета =
			"{""ok"":true,""already_exists"":true," +
			"""external_id"":""" + JsonEscape(ВнешнийИДЗаказа) + """," +
			"""doc_ref"":""" + JsonEscape(Строка(СсылкаСуществ)) + """," +
			"""doc_number"":""" + JsonEscape(Строка(НомерСуществ)) + """," +
			"""doc_date"":""" + JsonEscape(Формат(ДатаСуществ, "ДФ=yyyy-MM-dd HH:mm:ss")) + """," +
			"""deleted"":" + BoolJson(ПометкаУдаления) + "," +
			"""nav"":""" + JsonEscape(НавСсылка) + """}";

		Ответ = Новый HTTPСервисОтвет(200);
		Ответ.Заголовки["Content-Type"] = "application/json; charset=utf-8";
		Ответ.УстановитьТелоИзСтроки(ТелоОтвета);
		Возврат Ответ;

	КонецЕсли;

	// ---------------- CREATE DOCUMENT ----------------
	Док = Документы.ЗаказПокупателя.СоздатьДокумент();

	// дата
	Док.Дата = ТекущаяДата();
	ДатаСтрока = СокрЛП(Строка(JsonGet(ЗаказДанные, "date_insert", "")));
	Если НЕ ПустаяСтрока(ДатаСтрока) Тогда
		Попытка
			Док.Дата = Дата(Лев(ДатаСтрока, 19));
		Исключение
		КонецПопытки;
	КонецЕсли;

	// комментарий
	Попытка
		Док.Комментарий =
			"external_id: " + ВнешнийИДЗаказа + Символы.ПС +
			"Имя: " + Имя + Символы.ПС +
			"Email: " + Email + Символы.ПС +
			"Телефон: " + Телефон + Символы.ПС +
			"Адрес: " + Адрес;
	Исключение
	КонецПопытки;

	// Покупатель / О контрагенте / Состояние
	BackfillVisibilityFields(Док, PersonTypeId, PersonTypeName, Имя, Телефон, Email);

	// Отгрузка по умолчанию: через неделю от даты заказа
	BackfillShipmentFields(Док, Док.Дата);

	// обязательные поля для проведения
	ApplyUnfRequiredFields(Док, ВалютаКод, Курс, Кратность, ОперацияСтрока, НалогообложениеСтрока, ДатаПоступленияСтрока);

	// товары
	ТоварыМассив = JsonGet(ЗаказДанные, "items", Неопределено);
	Если ТоварыМассив = Неопределено Тогда
		Возврат ОтветОшибка(400, "no_items");
	КонецЕсли;

	ЕдШт = EnsureUnitSht();

	Для Каждого СтрокаТов Из ТоварыМассив Цикл

		Продукт = JsonGet(СтрокаТов, "product", Неопределено);

		ПВнешнИД = СокрЛП(Строка(JsonGet(Продукт, "external_id", "")));
		ПНаим = СокрЛП(Строка(JsonGet(Продукт, "name", "")));

		Кол = ToNumber(JsonGet(СтрокаТов, "quantity", 0));
		Цена = ToNumber(JsonGet(СтрокаТов, "price", 0));

		Номенклатура = ОбеспечитьНоменклатуру(ПВнешнИД, ПНаим);

		ТЧТовары = GetItemsTablePart(Док);
		СтрокаТЧ = ТЧТовары.Добавить();

		SetFieldIfExists(СтрокаТЧ, "Номенклатура", Номенклатура);
		SetFieldIfExists(СтрокаТЧ, "Количество", Кол);
		SetFieldIfExists(СтрокаТЧ, "Цена", Цена);
		SetFieldIfExists(СтрокаТЧ, "Сумма", Окр(Кол * Цена, 2));

		// Ед. = "штук/шт" всегда
		SetFieldIfExists(СтрокаТЧ, "Единица", ЕдШт);
		SetFieldIfExists(СтрокаТЧ, "ЕдиницаИзмерения", ЕдШт);
		SetFieldIfExists(СтрокаТЧ, "Ед", ЕдШт);
		SetFieldIfExists(СтрокаТЧ, "ЕдИзм", ЕдШт);

		// % НДС
		НДСПроцент = ToNumber(JsonGet(СтрокаТов, "vat_percent", JsonGet(СтрокаТов, "nds_percent", 0)));
		ApplyVatToLine(СтрокаТЧ, НДСПроцент);

	КонецЦикла;

	Попытка
		Док.Записать();
	Исключение
		Возврат ОтветОшибка(500, "write_failed: " + ОписаниеОшибки());
	КонецПопытки;

	НавСсылка = "";
	Попытка
		НавСсылка = ПолучитьНавигационнуюСсылку(Док.Ссылка);
	Исключение
	КонецПопытки;

	ТелоОтвета =
		"{""ok"":true," +
		"""external_id"":""" + JsonEscape(ВнешнийИДЗаказа) + """," +
		"""doc_ref"":""" + JsonEscape(Строка(Док.Ссылка)) + """," +
		"""doc_number"":""" + JsonEscape(Строка(Док.Номер)) + """," +
		"""doc_date"":""" + JsonEscape(Формат(Док.Дата, "ДФ=yyyy-MM-dd HH:mm:ss")) + """," +
		"""nav"":""" + JsonEscape(НавСсылка) + """}";

	Ответ = Новый HTTPСервисОтвет(200);
	Ответ.Заголовки["Content-Type"] = "application/json; charset=utf-8";
	Ответ.УстановитьТелоИзСтроки(ТелоОтвета);
	Возврат Ответ;

&НаСервере
Функция DebugModeFromRequest(Запрос) Экспорт
	Флаг = ВРег(СокрЛП(HeaderValue(Запрос, "X-MF-Debug")));
	Если (Флаг = "1") ИЛИ (Флаг = "Y") ИЛИ (Флаг = "YES") ИЛИ (Флаг = "TRUE") Тогда
		Возврат "META";
	КонецЕсли;
	Если (Флаг = "VALUES") ИЛИ (Флаг = "VAL") ИЛИ (Флаг = "V") ИЛИ (Флаг = "2") Тогда
		Возврат "VALUES";
	КонецЕсли;
	Возврат "";

	Зап.ЗаписатьКонецОбъекта();

	Зап.ЗаписатьИмяСвойства("requisites");

		Для Каждого Р Из МД.Реквизиты Цикл
			Зап.ЗаписатьНачалоОбъекта();

				Попытка Зап.ЗаписатьИмяСвойства("synonym"); Зап.ЗаписатьЗначение(Строка(РТЧ.Синоним)); Исключение Зап.ЗаписатьИмяСвойства("synonym"); Зап.ЗаписатьЗначение(""); КонецПопытки;
				Зап.ЗаписатьИмяСвойства("types"); WriteTypesToJson(Зап, РТЧ.Тип);

			КонецЦикла;
			Зап.ЗаписатьКонецМассива();

	Зап.ЗаписатьИмяСвойства("items_table_part_runtime");

	Попытка
		ТЧ = GetItemsTablePart(Док);
		// попробуем найти имя табличной части по метаданным
		Если МД <> Неопределено Тогда

				КонецЕсли;

		КонецЕсли;
	Исключение
	КонецПопытки;
	Зап.ЗаписатьЗначение(ИмяРантайм);

	Зап.ЗаписатьКонецОбъекта();

	ТелоОтвета = Зап.Закрыть();
	Ответ = Новый HTTPСервисОтвет(200);
	Ответ.Заголовки["Content-Type"] = "application/json; charset=utf-8";
	Ответ.УстановитьТелоИзСтроки(ТелоОтвета);
	Возврат Ответ;

				Счет = Счет + 1;
				Прервать;
			КонецЕсли;
		КонецЦикла;
	КонецЦикла;
	Возврат Счет;

Функция ResolveRefByTypeLabel(СинонимТипа, ТекстЗначения, AllowFirstFallback) Экспорт
	ТекстЗначения = СокрЛП(Строка(ТекстЗначения));
	Если ПустаяСтрока(СинонимТипа) Тогда Возврат Неопределено; КонецЕсли;

	ИмяПер = FindEnumNameBySynonym(СинонимТипа);
	Если НЕ ПустаяСтрока(ИмяПер) Тогда
		ЗнПер = EnumValueByText(ИмяПер, ТекстЗначения);
		Если ЗнПер <> Неопределено Тогда Возврат ЗнПер; КонецЕсли;
		Если AllowFirstFallback Тогда
			ЗнПер = EnumFirstValue(ИмяПер);
			Если ЗнПер <> Неопределено Тогда Возврат ЗнПер; КонецЕсли;
		КонецЕсли;
	КонецЕсли;

	ИмяСпр = FindCatalogNameBySynonym(СинонимТипа);
	Если НЕ ПустаяСтрока(ИмяСпр) Тогда
		Сс = CatalogRefByName(ИмяСпр, ТекстЗначения);
		Если ЗначениеЗаполнено(Сс) Тогда Возврат Сс; КонецЕсли;
		Если AllowFirstFallback Тогда
			Сс = CatalogFirstRef(ИмяСпр);
			Если ЗначениеЗаполнено(Сс) Тогда Возврат Сс; КонецЕсли;
		КонецЕсли;
	КонецЕсли;

	ИмяПВХ = FindPVHNameBySynonym(СинонимТипа);
	Если НЕ ПустаяСтрока(ИмяПВХ) Тогда
		Сс = PVHRefByName(ИмяПВХ, ТекстЗначения);
		Если ЗначениеЗаполнено(Сс) Тогда Возврат Сс; КонецЕсли;
		Если AllowFirstFallback Тогда
			Сс = PVHFirstRef(ИмяПВХ);
			Если ЗначениеЗаполнено(Сс) Тогда Возврат Сс; КонецЕсли;
		КонецЕсли;
	КонецЕсли;

	Возврат Неопределено;

Процедура SetVatRateToLineSmart(СтрокаТЧ, НДСПроцент) Экспорт
	ТипСтавки = GetRequisiteTypeName(СтрокаТЧ, "СтавкаНДС");
	Если ПустаяСтрока(ТипСтавки) Тогда ТипСтавки = "Ставка НДС"; КонецЕсли;

	Кандидаты = Новый Массив;
	Если НДСПроцент = 20 Тогда
		// В вашей УНФ ставки: "Общая", "Общая расчетная"

		Кандидаты.Добавить("Общая расчетная");
	ИначеЕсли НДСПроцент = 10 Тогда

		Кандидаты.Добавить("Без НДС");
	Иначе
		Кандидаты.Добавить("Без НДС");
	КонецЕсли;

	Ставка = Неопределено;
	Для Каждого К Из Кандидаты Цикл

		Ставка = ResolveRefByTypeLabel(ТипСтавки, "", Истина); // первый элемент
	КонецЕсли;

	Если Ставка <> Неопределено Тогда
		SetFieldIfExists(СтрокаТЧ, "СтавкаНДС", Ставка);
	КонецЕсли;

Функция JsonTryGet(Obj, Key, Значение) Экспорт
	Значение = Неопределено;

	Если Obj = Неопределено Тогда
		Возврат Ложь;
	КонецЕсли;

	Если ТипObj = Тип("Структура") ИЛИ ТипObj = Тип("ФиксированнаяСтруктура") Тогда
		Возврат Obj.Свойство(Key, Значение);
	КонецЕсли;

	Если ТипObj = Тип("Соответствие") ИЛИ ТипObj = Тип("ФиксированноеСоответствие") Тогда
		Для Каждого П Из Obj Цикл
			Если Строка(П.Ключ) = Строка(Key) Тогда
				Значение = П.Значение;
				Возврат Истина;
			КонецЕсли;
		КонецЦикла;
	КонецЕсли;

	Возврат Ложь;

			Если ЗначениеЗаполнено(В) Тогда Возврат В; КонецЕсли;
		КонецЕсли;

		Если ЗначениеЗаполнено(В) Тогда Возврат В; КонецЕсли;
	Исключение
		Возврат Неопределено;

Функция EnsureCurrencyForDoc(Док, Код) Экспорт
	Код = СокрЛП(Строка(Код));
	Если ПустаяСтрока(Код) Тогда Код = "RUB"; КонецЕсли;

	// 1) стандартный справочник Валюты
	Вал = EnsureCurrency(Код);
	Если ЗначениеЗаполнено(Вал) Тогда Возврат Вал; КонецЕсли;

	// 2) по типу реквизита документа (если валюта хранится в другом справочнике)
	ТипИмя = GetRequisiteTypeName(Док, "Валюта");
	Если Лев(ТипИмя, 11) = "Справочник" Тогда
		Позиция = Найти(ТипИмя, ".");
		Если Позиция > 0 Тогда
			ИмяСпр = Сред(ТипИмя, Позиция + 1);
			Вал = CatalogCurrencyRef(ИмяСпр, Код);
			Если ЗначениеЗаполнено(Вал) Тогда Возврат Вал; КонецЕсли;
		КонецЕсли;
	КонецЕсли;

	ТипИмя = GetRequisiteTypeName(Док, "ВалютаДокумента");
	Если Лев(ТипИмя, 11) = "Справочник" Тогда
		Позиция = Найти(ТипИмя, ".");
		Если Позиция > 0 Тогда
			ИмяСпр = Сред(ТипИмя, Позиция + 1);
			Вал = CatalogCurrencyRef(ИмяСпр, Код);
			Если ЗначениеЗаполнено(Вал) Тогда Возврат Вал; КонецЕсли;
		КонецЕсли;
	КонецЕсли;

	Возврат Неопределено;

&НаСервере
Процедура SetRequisiteByTextSmart(Док, ИмяРеквизита, Текст) Экспорт
	// 1) Попробуем присвоить строкой
	SetFieldIfExists(Док, ИмяРеквизита, Текст);

	// 2) Попробуем перечисление/справочник по типу реквизита
	ТипИмя = GetRequisiteTypeName(Док, ИмяРеквизита);
	Если ПустаяСтрока(ТипИмя) Тогда Возврат; КонецЕсли;

	Если Лев(ТипИмя, 15) = "Перечисление" Тогда
		Позиция = Найти(ТипИмя, ".");
		Если Позиция > 0 Тогда
			ИмяПеречисления = Сред(ТипИмя, Позиция + 1);
			ЗнПереч = EnumValueByText(ИмяПеречисления, Текст);
			Если ЗнПереч <> Неопределено Тогда
				SetFieldIfExists(Док, ИмяРеквизита, ЗнПереч);
			ИначеЕсли (ИмяРеквизита = "Состояние") ИЛИ (ИмяРеквизита = "Статус") Тогда
				ЗнПереч = EnumFirstValue(ИмяПеречисления);
				Если ЗнПереч <> Неопределено Тогда SetFieldIfExists(Док, ИмяРеквизита, ЗнПереч); КонецЕсли;
			КонецЕсли;
		КонецЕсли;
	ИначеЕсли Лев(ТипИмя, 11) = "Справочник" Тогда
		Позиция = Найти(ТипИмя, ".");
		Если Позиция > 0 Тогда
			ИмяСпр = Сред(ТипИмя, Позиция + 1);
			Сс = CatalogRefByName(ИмяСпр, Текст);
			Если НЕ ЗначениеЗаполнено(Сс) И ((ИмяРеквизита = "Состояние") ИЛИ (ИмяРеквизита = "Статус")) Тогда
				Сс = CatalogFirstRef(ИмяСпр);
			КонецЕсли;
			Если ЗначениеЗаполнено(Сс) Тогда SetFieldIfExists(Док, ИмяРеквизита, Сс); КонецЕсли;
		КонецЕсли;
	Иначе
		// В УНФ многие ссылочные типы отображаются как "Контрагент", "Валюта", "Ставка НДС" и т.п.
		// Пытаемся разрешить по синониму типа через Метаданные.
		AllowFirst = (ИмяРеквизита = "СостояниеЗаказа") ИЛИ (ИмяРеквизита = "ВидОперации") ИЛИ (ИмяРеквизита = "НалогообложениеНДС");
		Зн = ResolveRefByTypeLabel(ТипИмя, Текст, AllowFirst);
		Если Зн <> Неопределено Тогда
			SetFieldIfExists(Док, ИмяРеквизита, Зн);
		КонецЕсли;
	КонецЕсли;
КонецПроцедуры

Функция EnsurePaymentIndicatorRegisters(ЗаказСсылка, СуммаДокумента, ДатаПлатежа) Экспорт
	// Цель: чтобы в списках/формах УНФ появился индикатор оплаты (кружок),
	// который типовыми функциями считается через РС ФактОплатыЗаказов и РС ГрафикОплатыЗаказов.
	Если НЕ ЗначениеЗаполнено(ЗаказСсылка) Тогда Возврат Ложь; КонецЕсли;
	Если СуммаДокумента = Неопределено Тогда СуммаДокумента = 0; КонецЕсли;
	Если ДатаПлатежа = Неопределено Тогда ДатаПлатежа = ТекущаяДата(); КонецЕсли;

	Ок = Истина;

	// 1) Факт оплаты заказов: 0% оплаты, но сумма заказа известна
	Попытка
		НаборФакт = РегистрыСведений.ФактОплатыЗаказов.СоздатьНаборЗаписей();
		Попытка НаборФакт.Отбор.СчетНаОплату.Установить(ЗаказСсылка); Исключение КонецПопытки;
		НаборФакт.Прочитать();
		Попытка НаборФакт.Очистить(); Исключение КонецПопытки;
		ЗапФ = НаборФакт.Добавить();
		Попытка ЗапФ.Период = ТекущаяДата(); Исключение КонецПопытки;
		Попытка ЗапФ.СчетНаОплату = ЗаказСсылка; Исключение КонецПопытки;
		Попытка ЗапФ.Сумма = СуммаДокумента; Исключение КонецПопытки;
		Попытка ЗапФ.СуммаОплаты = 0; Исключение КонецПопытки;
		Попытка ЗапФ.СуммаАванса = 0; Исключение КонецПопытки;
		НаборФакт.Записать();
	Исключение
		Ок = Ложь;
	КонецПопытки;

	// 2) График оплаты заказов: плановая сумма на плановую дату (для "ждём сегодня/не просрочена")
	Попытка
		НаборГр = РегистрыСведений.ГрафикОплатыЗаказов.СоздатьНаборЗаписей();
		Попытка НаборГр.Отбор.СчетНаОплату.Установить(ЗаказСсылка); Исключение КонецПопытки;
		НаборГр.Прочитать();
		Попытка НаборГр.Очистить(); Исключение КонецПопытки;
		ЗапГ = НаборГр.Добавить();
		Попытка ЗапГ.Период = НачалоДня(ДатаПлатежа); Исключение Попытка ЗапГ.Период = ДатаПлатежа; Исключение КонецПопытки; КонецПопытки;
		Попытка ЗапГ.СчетНаОплату = ЗаказСсылка; Исключение КонецПопытки;
		Попытка ЗапГ.Сумма = СуммаДокумента; Исключение КонецПопытки;
		НаборГр.Записать();
	Исключение
		Ок = Ложь;
	КонецПопытки;

	Возврат Ок;
КонецФункции

Функция BuildPaymentTabDiagJson(Док) Экспорт
	Заплан = Ложь;
	ПК = 0;
	ПР = 0;
	ЛучИмяТЧ = "";
	ЛучКол = 0;
	RowDiag = "";
	RowDiag2 = "";
	RowDump = "";
	RegDiag = "";
	FactCnt = -1;
	PlanCnt = -1;
	Если Док <> Неопределено Тогда
		Попытка Заплан = GetFieldValueSafe(Док, "ЗапланироватьОплату"); Исключение Заплан = Ложь; КонецПопытки;
		Попытка
			ТЧ = TryGetTablePart(Док, "ПлатежныйКалендарь");
			Если ТЧ <> Неопределено Тогда ПК = ТЧ.Количество(); КонецЕсли;
		Исключение
		КонецПопытки;
		Попытка
			ТЧ2 = TryGetTablePart(Док, "Предоплата");
			Если ТЧ2 <> Неопределено Тогда ПР = ТЧ2.Количество(); КонецЕсли;
		Исключение
		КонецПопытки;

		// Какая табличная часть оплаты была найдена общим резолвером
		Попытка
			ТЧЛ = GetPaymentPlanTablePart(Док);
			Если ТЧЛ <> Неопределено Тогда
				ЛучКол = ТЧЛ.Количество();
				// Попробуем определить имя табчасти через метаданные
				Попытка
					МД = Док.Метаданные();
					Если МД <> Неопределено Тогда
						Для Каждого ТЧМ Из МД.ТабличныеЧасти Цикл
							Имя0 = "";
							Попытка Имя0 = Строка(ТЧМ.Имя); Исключение Имя0 = ""; КонецПопытки;
							Если НЕ ПустаяСтрока(Имя0) Тогда
								ТЧX = TryGetTablePart(Док, Имя0);
								Если ТЧX = ТЧЛ Тогда
									ЛучИмяТЧ = Имя0;
									Прервать;
								КонецЕсли;
							КонецЕсли;
						КонецЦикла;
					КонецЕсли;
				Исключение
				КонецПопытки;
			КонецЕсли;
		Исключение
		КонецПопытки;

		// Диагностика ключевых полей первой строки платежного календаря
		Попытка
			ТЧПК = TryGetTablePart(Док, "ПлатежныйКалендарь");
			Если (ТЧПК <> Неопределено) И (ТЧПК.Количество() > 0) Тогда
				Стр0 = ТЧПК[0];
				Sum1 = ToNumber(GetFieldValueSafe(Стр0, "СуммаОплаты"));
				Sum2 = ToNumber(GetFieldValueSafe(Стр0, "Сумма"));
				Sum3 = ToNumber(GetFieldValueSafe(Стр0, "СуммаКОплате"));
				Dat1 = GetFieldValueSafe(Стр0, "ДатаОплаты");
				Dat2 = GetFieldValueSafe(Стр0, "ДатаПлатежа");
				Dat3 = GetFieldValueSafe(Стр0, "Дата");
				RowDiag =
					"{""СуммаОплаты"":" + JsonNumberLiteral(Sum1) +
					",""Сумма"":" + JsonNumberLiteral(Sum2) +
					",""СуммаКОплате"":" + JsonNumberLiteral(Sum3) +
					",""ДатаОплаты"":""" + JsonEscape(DateYmdSafe(Dat1)) + """" +
					",""ДатаПлатежа"":""" + JsonEscape(DateYmdSafe(Dat2)) + """" +
					",""Дата"":""" + JsonEscape(DateYmdSafe(Dat3)) + """" +
					"}";
				RowDump = DumpTabRowFieldsJsonByDocMetadata(Док, "ПлатежныйКалендарь", Стр0);
			КонецЕсли;
		Исключение
		КонецПопытки;

		// Диагностика реквизитов документа, которые могут влиять на индикатор оплаты
		Попытка
			D1 = GetFieldValueSafe(Док, "ДатаОплаты");
			D2 = GetFieldValueSafe(Док, "СрокОплаты");
			D3 = GetFieldValueSafe(Док, "ДатаПлатежа");
			RowDiag2 =
				"{""ДатаОплаты"":""" + JsonEscape(DateYmdSafe(D1)) + """" +
				",""СрокОплаты"":""" + JsonEscape(DateYmdSafe(D2)) + """" +
				",""ДатаПлатежа"":""" + JsonEscape(DateYmdSafe(D3)) + """" +
				"}";
		Исключение
		КонецПопытки;

		// Диагностика регистра "ПлатежныйКалендарь": есть ли записи, по которым UI может рисовать кружок
		Попытка
			СсДок = Док.Ссылка;
			RegDiag = PaymentCalendarRegisterDiagJson(СсДок);
		Исключение
			RegDiag = "{}";
		КонецПопытки;

		// РС ФактОплатыЗаказов / ГрафикОплатыЗаказов (для индикатора оплаты)
		Попытка
			СсДок2 = Док.Ссылка;
			НФ = РегистрыСведений.ФактОплатыЗаказов.СоздатьНаборЗаписей();
			Попытка НФ.Отбор.СчетНаОплату.Установить(СсДок2); Исключение КонецПопытки;
			НФ.Прочитать();
			FactCnt = НФ.Количество();
		Исключение
		КонецПопытки;
		Попытка
			СсДок3 = Док.Ссылка;
			НГ = РегистрыСведений.ГрафикОплатыЗаказов.СоздатьНаборЗаписей();
			Попытка НГ.Отбор.СчетНаОплату.Установить(СсДок3); Исключение КонецПопытки;
			НГ.Прочитать();
			PlanCnt = НГ.Количество();
		Исключение
		КонецПопытки;
	КонецЕсли;

	Если ПустаяСтрока(RowDiag) Тогда RowDiag = "{}"; КонецЕсли;
	Если ПустаяСтрока(RowDiag2) Тогда RowDiag2 = "{}"; КонецЕсли;
	Если ПустаяСтрока(RowDump) Тогда RowDump = "{}"; КонецЕсли;
	Если ПустаяСтрока(RegDiag) Тогда RegDiag = "{}"; КонецЕсли;

	Возврат "{""ЗапланироватьОплату"":" + BoolJson(Заплан) +
		",""ПлатежныйКалендарь_count"":" + Строка(ПК) +
		",""Предоплата_count"":" + Строка(ПР) +
		",""resolved_table_part"":""" + JsonEscape(ЛучИмяТЧ) + """" +
		",""resolved_count"":" + Строка(ЛучКол) +
		",""ПлатежныйКалендарь_row0"":" + RowDiag +
		",""ПлатежныйКалендарь_row0_dump"":" + RowDump +
		",""doc_due_fields"":" + RowDiag2 +
		",""paycal_reg_diag"":" + RegDiag +
		",""factpay_count"":" + Строка(FactCnt) +
		",""payplan_count"":" + Строка(PlanCnt) +
		"}";

	Мас.Добавить("PHONE_MOBILE");

				S2 = СокрЛП(Строка(П.Значение));

					Возврат S2;

		КонецЦикла;
	КонецЕсли;

// Число для встраивания в JSON: без пробелов/NBSP и без ведущих нулей у целой части (иначе json_decode в PHP падает).
&НаСервере
Функция JsonNumberLiteral(ЗначениеЧисла) Экспорт
	Ч = ToNumber(ЗначениеЧисла);
	Попытка
		S = "";
		Попытка
			S = Формат(Ч, "ЧРГ=; ЧРД=.; ЧН=0; ЧВН=; ЧЦ=24; ЧДЦ=8");
		Исключение
			S = Строка(Ч);
		КонецПопытки;
		S = СокрЛП(S);
		// Убираем любые пробельные разделители групп (в т.ч. узкий неразрывный U+202F из ru_RU).
		S = СтрЗаменить(S, " ", "");
		S = СтрЗаменить(S, Символы.НПП, "");
		S = СтрЗаменить(S, Символ(160), "");
		S = СтрЗаменить(S, Символ(8239), "");
		S = СтрЗаменить(S, ",", ".");
		// JSON не допускает «000001» у целой части — нормализуем.
		Минус = Ложь;
		Если СтрДлина(S) > 0 И Лев(S, 1) = "-" Тогда
			Минус = Истина;
			S = Сред(S, 2);
		КонецЕсли;
		ПозТчк = СтрНайти(S, ".");
		ЦелСтр = ?(ПозТчк > 0, Лев(S, ПозТчк - 1), S);
		ДробСтр = ?(ПозТчк > 0, Сред(S, ПозТчк + 1), "");
		Пока СтрДлина(ЦелСтр) > 1 И Лев(ЦелСтр, 1) = "0" Цикл
			ЦелСтр = Сред(ЦелСтр, 2);
		КонецЦикла;
		Если ПустаяСтрока(ЦелСтр) Тогда ЦелСтр = "0"; КонецЕсли;
		S = ЦелСтр;
		Если НЕ ПустаяСтрока(ДробСтр) Тогда
			S = S + "." + ДробСтр;
		КонецЕсли;
		Если Минус Тогда S = "-" + S; КонецЕсли;
		Возврат S;
	Исключение
		Возврат "0";
	КонецПопытки;

Процедура EnsureCurrencyRateNotZero(Валюта, Момент) Экспорт
	Если НЕ ЗначениеЗаполнено(Валюта) Тогда Возврат; КонецЕсли;
	Если Момент = Неопределено Тогда Момент = ТекущаяДата(); КонецЕсли;

	// Важно: РС "КурсыВалют" периодичность = День и допускает несколько строк на один день (LineNumber).
	// СрезПоследних выбирает "последнюю" по (Период + LineNumber). Поэтому правим ВСЕ строки
	// на максимальный Период <= Момент, иначе можно исправить не ту строку и ноль останется.
	Попытка
		Набор = РегистрыСведений.КурсыВалют.СоздатьНаборЗаписей();
		Попытка Набор.Отбор.Валюта.Установить(Валюта); Исключение КонецПопытки;
		Попытка Набор.Прочитать(); Исключение КонецПопытки;

		МаксПериод = Дата("00010101");
		Для Каждого S Из Набор Цикл
			П = Неопределено;
			Попытка П = S.Период; Исключение П = Неопределено; КонецПопытки;
			Если П = Неопределено Тогда Продолжить; КонецЕсли;
			Если П <= Момент И П > МаксПериод Тогда
				МаксПериод = П;
			КонецЕсли;
		КонецЦикла;

		Если МаксПериод = Дата("00010101") Тогда
			// нет записей — добавим одну на текущий день
			Стр = Набор.Добавить();
			// периодичность День — ставим начало дня
			Стр.Период = НачалоДня(Момент);
			Стр.Валюта = Валюта;
			Стр.Курс = 1;
			Стр.Кратность = 1;
			Набор.Записать();
		Иначе
			Исправили = Ложь;
			Для Каждого S2 Из Набор Цикл
				П2 = Неопределено;
				Попытка П2 = S2.Период; Исключение П2 = Неопределено; КонецПопытки;
				Если П2 <> МаксПериод Тогда Продолжить; КонецЕсли;
				Курс2 = 0; Кратн2 = 0;
				Попытка Курс2 = ToNumber(S2.Курс); Исключение Курс2 = 0; КонецПопытки;
				Попытка Кратн2 = ToNumber(S2.Кратность); Исключение Кратн2 = 0; КонецПопытки;
				Если (Курс2 <= 0) ИЛИ (Кратн2 <= 0) Тогда
					S2.Курс = 1;
					S2.Кратность = 1;
					Исправили = Истина;
				КонецЕсли;
			КонецЦикла;
			Если Исправили Тогда
				Набор.Записать();
			КонецЕсли;
		КонецЕсли;
	Исключение
	КонецПопытки;

&НаСервере
Функция ToNumber(Значение) Экспорт
	Если Значение = Неопределено Тогда
		Возврат 0;
	КонецЕсли;

	// Если уже число — возвращаем как есть
	Попытка
		Если ТипЗнч(Значение) = Тип("Число") Тогда
			Возврат Значение;
		КонецЕсли;
	Исключение
	КонецПопытки;

	// Строки нормализуем ВСЕГДА (иначе Число("1 234,56") часто "успешно" возвращает 1)
	Стр = СокрЛП(Строка(Значение));
	Если ПустаяСтрока(Стр) Тогда Возврат 0; КонецЕсли;

	Рез = "";
	БылаТочка = Ложь;
	Для Индекс = 1 По СтрДлина(Стр) Цикл
		Симв = Сред(Стр, Индекс, 1);
		Если (Симв >= "0") И (Симв <= "9") Тогда
			Рез = Рез + Симв;
		ИначеЕсли (Симв = "-") И ПустаяСтрока(Рез) Тогда
			Рез = "-";
		ИначеЕсли ((Симв = ",") ИЛИ (Симв = ".")) И (НЕ БылаТочка) Тогда
			Рез = Рез + ".";
			БылаТочка = Истина;
		Иначе
			// пропускаем (пробелы, валюты, буквы и т.п.)
		КонецЕсли;
	КонецЦикла;

	Если ПустаяСтрока(Рез) ИЛИ (Рез = "-") Тогда Возврат 0; КонецЕсли;
	Попытка
		Возврат Число(Рез);
	Исключение
		// последний шанс
		Попытка
			Возврат Число(Значение);
		Исключение
			Возврат 0;
		КонецПопытки;
	КонецПопытки;

Процедура ApplyPaymentFromJson(Док, ЗаказДанные) Экспорт
	// Цель: заполнить минимальные поля, чтобы в форме "Заказ покупателя"
	// появилась/заполнилась вкладка "Оплата" (план/график оплат).
	// ВАЖНО: тут НЕ создаём и НЕ проводим документы оплаты.
	Попытка
		// ВАЖНО: вкладка "Оплата" должна появляться уже при создании заказа,
		// поэтому НЕ выходим, даже если блока payment нет — берём сумму из order.price.
		Оплата = JsonGet(ЗаказДанные, "payment", Неопределено);
		План = Неопределено;
		Инфо = Неопределено;
		Если Оплата <> Неопределено Тогда
			План = JsonGet(Оплата, "plan", Неопределено);
			Инфо = JsonGet(Оплата, "info", Неопределено);
		КонецЕсли;
		Событие = JsonGet(ЗаказДанные, "payment_event", Неопределено);

		СуммаПлана = 0;
		Если План <> Неопределено Тогда
			СуммаПлана = ToNumber(JsonGet(План, "sum", 0));
		КонецЕсли;
		Если СуммаПлана <= 0 И Инфо <> Неопределено Тогда
			СуммаПлана = ToNumber(JsonGet(Инфо, "sum", 0));
		КонецЕсли;
		Если СуммаПлана <= 0 Тогда
			СуммаПлана = ToNumber(JsonGet(ЗаказДанные, "price", 0));
		КонецЕсли;
		Если СуммаПлана < 0 Тогда СуммаПлана = 0; КонецЕсли;

		ДатаПлатежаСтрока = "";
		Если План <> Неопределено Тогда
			ДатаПлатежаСтрока = СокрЛП(Строка(JsonGet(План, "due_at", "")));
		КонецЕсли;
		Если ПустаяСтрока(ДатаПлатежаСтрока) Тогда
			ДатаПлатежаСтрока = СокрЛП(Строка(JsonGet(ЗаказДанные, "date_insert", "")));
		КонецЕсли;
		ДатаПлатежа = ParseDateSafe(ДатаПлатежаСтрока, Док.Дата);
		// ВАЖНО для индикатора оплаты в заказе:
		// если плановая дата оплаты = сегодня, УНФ показывает жёлтый статус "Без оплаты (ждём сегодня)".
		// Нам нужен зелёный "Без оплаты (не просрочена)" — значит плановая дата должна быть в будущем.
		Попытка
			Сегодня0 = НачалоДня(ТекущаяДата());
			План0 = НачалоДня(ДатаПлатежа);
			Если План0 <= Сегодня0 Тогда
				ДатаПлатежа = Сегодня0 + 24 * 60 * 60; // завтра
			КонецЕсли;
		Исключение
		КонецПопытки;

		// В некоторых УНФ индикатор "Оплата" завязан не только на табличную часть,
		// но и на реквизиты документа со сроком/датой оплаты.
		SetFieldIfExists(Док, "ДатаОплаты", ДатаПлатежа);
		SetFieldIfExists(Док, "СрокОплаты", ДатаПлатежа);
		SetFieldIfExists(Док, "ДатаПлатежа", ДатаПлатежа);
		SetFieldIfExists(Док, "ПлановаяДата", ДатаПлатежа);
		SetFieldIfExists(Док, "ПлановаяДатаОплаты", ДатаПлатежа);
		SetFieldIfExists(Док, "ПлановаяДатаПлатежа", ДатаПлатежа);

		// В вашей УНФ есть явный реквизит "Запланировать оплату" — он влияет на вкладку/план оплат.
		SetFieldIfExists(Док, "ЗапланироватьОплату", Истина);

		// Полезно проставить идентификатор платежа (если есть UUID PayKeeper / voucher).
		ИдПлатежа = "";
		Если (Инфо <> Неопределено) Тогда
			ИдПлатежа = СокрЛП(Строка(JsonGet(Инфо, "ps_invoice_id", "")));
			Если ПустаяСтрока(ИдПлатежа) Тогда
				ИдПлатежа = СокрЛП(Строка(JsonGet(Инфо, "pay_voucher_num", "")));
			КонецЕсли;
		КонецЕсли;
		Если ПустаяСтрока(ИдПлатежа) И (Событие <> Неопределено) Тогда
			ИдПлатежа = СокрЛП(Строка(JsonGet(Событие, "ps_invoice_id", "")));
		КонецЕсли;
		Если НЕ ПустаяСтрока(ИдПлатежа) Тогда
			SetFieldIfExists(Док, "ИдентификаторПлатежа", ИдПлатежа);
		КонецЕсли;

		// КРИТИЧНО для вашей конфигурации:
		// стандартное проведение УНФ в запросах делит на Предоплата.Кратность; если там 0 — получаем division by zero.
		// Поэтому НЕ заполняем табчасть "Предоплата" и принудительно очищаем её (если есть),
		// оставляя только "ПлатежныйКалендарь" для UI.
		FillPaymentTableIfExists(Док, "ПлатежныйКалендарь", СуммаПлана, ДатаПлатежа);
		ТЧПред = TryGetTablePart(Док, "Предоплата");
		Если ТЧПред <> Неопределено Тогда
			Попытка ТЧПред.Очистить(); Исключение КонецПопытки;
		КонецЕсли;

		// И как fallback — ту табчасть, которую нашёл общий резолвер (НО не "Предоплата").
		ТЧПлан = GetPaymentPlanTablePart(Док);
		Если (ТЧПлан <> Неопределено) И (ТЧПред = Неопределено ИЛИ ТЧПлан <> ТЧПред) Тогда
			Попытка ТЧПлан.Очистить(); Исключение КонецПопытки;
			Стр = ТЧПлан.Добавить();
			SetFieldIfExists(Стр, "Сумма", СуммаПлана);
			SetFieldIfExists(Стр, "СуммаПлатежа", СуммаПлана);
			SetFieldIfExists(Стр, "СуммаКОплате", СуммаПлана);
			SetFieldIfExists(Стр, "СуммаОплаты", СуммаПлана);
			SetFieldIfExists(Стр, "СуммаРасчетов", СуммаПлана);
			SetFieldIfExists(Стр, "СуммаНДСОплаты", 0);
			SetFieldIfExists(Стр, "ПроцентОплаты", 100);
			SetFieldIfExists(Стр, "ДатаПлатежа", ДатаПлатежа);
			SetFieldIfExists(Стр, "Дата", ДатаПлатежа);
			SetFieldIfExists(Стр, "ПлановаяДата", ДатаПлатежа);
			SetFieldIfExists(Стр, "СрокОплаты", ДатаПлатежа);
			SetFieldIfExists(Стр, "ДатаОплаты", ДатаПлатежа);
		КонецЕсли;

		// В некоторых УНФ вкладка "Оплата" дополнительно зависит от реквизита "Условия оплаты" / "Вариант оплаты".
		// Пробуем проставить предоплату, если такие реквизиты есть (ошибки/несовпадения проглотятся).
		SetRequisiteBySynonymSmart(Док, "Условия оплаты", "Предоплата");
		SetRequisiteBySynonymSmart(Док, "Условия оплаты", "Полная предоплата");
		SetRequisiteBySynonymSmart(Док, "Вариант оплаты", "Предоплата");
		SetRequisiteBySynonymSmart(Док, "Порядок оплаты", "Предоплата");

		// Индикатор оплаты (кружок) в вашей УНФ считается через регистры сведений.
		// Заполняем их явно, чтобы кружок появлялся сразу после создания заказа.
		Попытка
			СуммаДок0 = ToNumber(GetFieldValueSafe(Док, "СуммаДокумента"));
			Если СуммаДок0 <= 0 Тогда
				СуммаДок0 = СуммаПлана;
			КонецЕсли;
			EnsurePaymentIndicatorRegisters(Док.Ссылка, СуммаДок0, ДатаПлатежа);
		Исключение
		КонецПопытки;

	Исключение
		// Безопасно: если табличной части/полей нет — просто не заполним.
	КонецПопытки;

Процедура FillPaymentTableIfExists(Док, ИмяТЧ, СуммаПлана, ДатаПлатежа) Экспорт
	ТЧ = TryGetTablePart(Док, ИмяТЧ);
	Если ТЧ = Неопределено Тогда Возврат; КонецЕсли;

	// Идемпотентно: всегда приводим к одной строке
	Попытка
		Попытка ТЧ.Очистить(); Исключение КонецПопытки;
		Стр = ТЧ.Добавить();
	Исключение
		// если табчасть/строки в вашей конфигурации не поддерживают добавление — просто пропустим
		Возврат;
	КонецПопытки;

	// Поля "ПлатежныйКалендарь" (в разных УНФ имена могут отличаться)
	// Даты
	SetFieldIfExists(Стр, "ДатаОплаты", ДатаПлатежа);
	SetFieldIfExists(Стр, "ДатаПлатежа", ДатаПлатежа);
	SetFieldIfExists(Стр, "Дата", ДатаПлатежа);
	SetFieldIfExists(Стр, "ПлановаяДата", ДатаПлатежа);
	SetFieldIfExists(Стр, "СрокОплаты", ДатаПлатежа);
	// Суммы
	SetFieldIfExists(Стр, "СуммаОплаты", СуммаПлана);
	SetFieldIfExists(Стр, "Сумма", СуммаПлана);
	SetFieldIfExists(Стр, "СуммаКОплате", СуммаПлана);
	SetFieldIfExists(Стр, "СуммаПлатежа", СуммаПлана);
	SetFieldIfExists(Стр, "СуммаРасчетов", СуммаПлана);
	SetFieldIfExists(Стр, "ПроцентОплаты", 100);
	SetFieldIfExists(Стр, "СуммаНДСОплаты", 0);

	// В вашей конфигурации в табличных частях оплаты есть Курс/Кратность,
	// и стандартные запросы проведения делят на Кратность — если там 0, получаем division by zero.
	Курс0 = ToNumber(GetFieldValueSafe(Док, "Курс"));
	Если Курс0 <= 0 Тогда Курс0 = 1; КонецЕсли;
	Кратн0 = ToNumber(GetFieldValueSafe(Док, "Кратность"));
	Если Кратн0 <= 0 Тогда Кратн0 = 1; КонецЕсли;
	SetFieldIfExists(Стр, "Курс", Курс0);
	SetFieldIfExists(Стр, "КурсРасчетов", Курс0);
	SetFieldIfExists(Стр, "КурсВалюты", Курс0);
	SetFieldIfExists(Стр, "Кратность", Кратн0);
	SetFieldIfExists(Стр, "КратностьРасчетов", Кратн0);
	SetFieldIfExists(Стр, "КратностьВалюты", Кратн0);

	// Поля "Предоплата" и общие варианты
	SetFieldIfExists(Стр, "СуммаПлатежа", СуммаПлана);
	SetFieldIfExists(Стр, "СуммаРасчетов", СуммаПлана);
	// Часто обязательные булевы поля в табчасти "Предоплата"
	Если ИмяТЧ = "Предоплата" Тогда
		SetFieldIfExists(Стр, "ОплатаСертификатом", Ложь);
		SetFieldIfExists(Стр, "ОплатаБонусами", Ложь);
		SetFieldIfExists(Стр, "ЭтоПредоплатаБезЗаказа", Ложь);
	КонецЕсли;
	SetFieldIfExists(Стр, "Сумма", СуммаПлана);
	SetFieldIfExists(Стр, "ДатаПлатежа", ДатаПлатежа);
	SetFieldIfExists(Стр, "Дата", ДатаПлатежа);

Функция ResolveCounterpartyKind(PersonTypeId, PersonTypeName, BuyerName) Экспорт
	// Просили строго: "Юр. лицо" или "Физ. лицо"
	Если PersonTypeId > 1 Тогда
		Возврат "Юр. лицо";
	КонецЕсли;

	Если Найти(ВРег(Строка(PersonTypeName)), "ЮР") > 0 Тогда
		Возврат "Юр. лицо";
	КонецЕсли;

	ИмяВРег = ВРег(Строка(BuyerName));
	Если (Найти(ИмяВРег, "ООО") > 0) ИЛИ (Найти(ИмяВРег, "АО") > 0) ИЛИ (Найти(ИмяВРег, "ЗАО") > 0) ИЛИ (Найти(ИмяВРег, "ОАО") > 0) ИЛИ (Найти(ИмяВРег, "ПАО") > 0) ИЛИ (Найти(ИмяВРег, "ИП") > 0) Тогда
		Возврат "Юр. лицо";
	КонецЕсли;

	Возврат "Физ. лицо";

&НаСервере
Процедура CounterpartyCIUpsert(ТЧ, Представление, ИмяТипаПеречисления, ИменаВидов, ПодстрПоискВида) Экспорт
	// УНФ 3 / БСП: телефон, e-mail и адреса в карточке — из ТЧ «КонтактнаяИнформация».
	Если ТЧ = Неопределено ИЛИ ПустаяСтрока(Представление) Тогда
		Возврат;
	КонецЕсли;

	ТипЗн = Неопределено;
	Попытка
		Выполнить("ТипЗн = Перечисления.ТипыКонтактнойИнформации." + ИмяТипаПеречисления + ";");
	Исключение
		ТипЗн = Неопределено;
	КонецПопытки;

	ВидЗн = CounterpartyCIKindRef(ИменаВидов, ПодстрПоискВида);

	ИндНайден = -1;
	Попытка
		Для Индекс = 0 По ТЧ.Количество() - 1 Цикл
			Стр = ТЧ[Индекс];
			Совп = Ложь;
			Если ТипЗн <> Неопределено Тогда
				Попытка
					Если Стр.Тип = ТипЗн Тогда
						Совп = Истина;
					КонецЕсли;
				Исключение
				КонецПопытки;
			КонецЕсли;
			Если (НЕ Совп) И ЗначениеЗаполнено(ВидЗн) Тогда
				Попытка
					Если Стр.Вид = ВидЗн Тогда
						Совп = Истина;
					КонецЕсли;
				Исключение
				КонецПопытки;
			КонецЕсли;
			Если Совп Тогда
				ИндНайден = Индекс;
				Прервать;
			КонецЕсли;
		КонецЦикла;
	Исключение
	КонецПопытки;

	Попытка
		Если ИндНайден >= 0 Тогда
			Стр2 = ТЧ[ИндНайден];
			Стр2.Представление = Представление;
			Если ТипЗн <> Неопределено Тогда
				Стр2.Тип = ТипЗн;
			КонецЕсли;
			Если ЗначениеЗаполнено(ВидЗн) Тогда
				Стр2.Вид = ВидЗн;
			КонецЕсли;
			CounterpartyCIFillValueJSON(Стр2, Представление, ТипЗн);
			SetFieldIfExists(Стр2, "ДействуетС", НачалоДня(ТекущаяДата()));
		Иначе
			Нстр = ТЧ.Добавить();
			Если ТипЗн <> Неопределено Тогда
				Нстр.Тип = ТипЗн;
			КонецЕсли;
			Нстр.Представление = Представление;
			Если ЗначениеЗаполнено(ВидЗн) Тогда
				Нстр.Вид = ВидЗн;
			КонецЕсли;
			CounterpartyCIFillValueJSON(Нстр, Представление, ТипЗн);
			SetFieldIfExists(Нстр, "ДействуетС", НачалоДня(ТекущаяДата()));
		КонецЕсли;
	Исключение
	КонецПопытки;

Функция LikePatternFromNormalized(Норм) Экспорт
	Норм = СокрЛП(Строка(Норм));
	Если ПустаяСтрока(Норм) Тогда Возврат "%"; КонецЕсли;
	// Паттерн вида %A%B%1%2%3% — чтобы совпало даже если в артикуле есть пробелы/тире
	Если СтрДлина(Норм) > 64 Тогда
		Норм = Лев(Норм, 64);
	КонецЕсли;
	Пат = "%";
	Для Индекс = 1 По СтрДлина(Норм) Цикл
		Пат = Пат + Сред(Норм, Индекс, 1) + "%";
	КонецЦикла;
	Возврат Пат;

Функция FindNomenclatureByArticle(Артикул) Экспорт
	Артикул0 = СокрЛП(Строка(Артикул));
	Норм = NormalizeArticle(Артикул0);
	Если ПустаяСтрока(Норм) Тогда Возврат Неопределено; КонецЕсли;

	// 1) Быстрый путь: точное совпадение по Артикул/Код
	Попытка
		Запрос = Новый Запрос;
		Запрос.Текст =
		"ВЫБРАТЬ ПЕРВЫЕ 1
		|	Номенклатура.Ссылка КАК Ссылка
		|ИЗ Справочник.Номенклатура КАК Номенклатура
		|ГДЕ Номенклатура.ПометкаУдаления = ЛОЖЬ
		|  И (Номенклатура.Артикул = &А ИЛИ Номенклатура.Код = &А)";
		Запрос.УстановитьПараметр("А", Артикул0);
		Рез = Запрос.Выполнить().Выбрать();
		Если Рез.Следующий() Тогда Возврат Рез.Ссылка; КонецЕсли;
	Исключение
	КонецПопытки;

	// 2) Толерантный поиск: ПОДОБНО + последующая нормализация кандидатов
	Пат = LikePatternFromNormalized(Норм);
	Попытка
		Запрос2 = Новый Запрос;
		Запрос2.Текст =
		"ВЫБРАТЬ ПЕРВЫЕ 200
		|	Ном.Ссылка КАК Ссылка,
		|	Ном.Артикул КАК Артикул,
		|	Ном.Код КАК Код
		|ИЗ Справочник.Номенклатура КАК Ном
		|ГДЕ Ном.ПометкаУдаления = ЛОЖЬ
		|  И (Ном.Артикул ПОДОБНО &П ИЛИ Ном.Код ПОДОБНО &П)";
		Запрос2.УстановитьПараметр("П", Пат);
		В = Запрос2.Выполнить().Выбрать();
		Пока В.Следующий() Цикл
			Н1 = NormalizeArticle(В.Артикул);
			Н2 = NormalizeArticle(В.Код);
			Если (Н1 = Норм) ИЛИ (Н2 = Норм) Тогда
				Возврат В.Ссылка;
			КонецЕсли;
		КонецЦикла;
	Исключение
	КонецПопытки;

	Возврат Неопределено;

Функция ОбеспечитьНоменклатуру(Артикул, Наименование, Бренд, ЕдШт) Экспорт

	Артикул = СокрЛП(Строка(Артикул));
	Если ПустаяСтрока(Артикул) Тогда
		Артикул = "bitrix-no-article";
	КонецЕсли;

	// 1) Сначала ищем существующую номенклатуру по артикулу (толерантно)
	Найдено = FindNomenclatureByArticle(Артикул);
	Если Найдено <> Неопределено Тогда
		Возврат Найдено;
	КонецЕсли;

	// 2) Не нашли — создаём как ТОВАР (Запас), а не услугу
	Эл = Справочники.Номенклатура.СоздатьЭлемент();

	ИмяНом = СокрЛП(Строка(Наименование));
	Если ПустаяСтрока(ИмяНом) Тогда
		ИмяНом = Артикул;
	КонецЕсли;

	Попытка Эл.Наименование = ИмяНом; Исключение КонецПопытки;
	// Наименование для печати — в разных УНФ может отличаться имя реквизита
	SetFieldIfExists(Эл, "НаименованиеДляПечати", ИмяНом);
	SetFieldIfExists(Эл, "НаименованиеДляПечатиПолное", ИмяНом);
	SetFieldIfExists(Эл, "ПолноеНаименование", ИмяНом);

	// Артикул + Код = артикул (как просили). Если код ограничен по длине — SetFieldIfExists не уронит создание.
	Попытка Эл.Артикул = Артикул; Исключение КонецПопытки;
	SetFieldIfExists(Эл, "Код", Артикул);

	// Тип = Запас (товар)
	Попытка
		Эл.ТипНоменклатуры = Перечисления.ТипыНоменклатуры.Запас;
	Исключение
		// fallback на возможное поле "Тип"
		SetFieldIfExists(Эл, "Тип", Перечисления.ТипыНоменклатуры.Запас);
	КонецПопытки;

	// Категория = бренд
	Кат = EnsureNomenclatureCategoryByName(Бренд);
	Если Кат <> Неопределено Тогда
		SetFieldIfExists(Эл, "КатегорияНоменклатуры", Кат);
		SetFieldIfExists(Эл, "Категория", Кат);
	КонецЕсли;

	// Ед. изм. хранения = "шт"
	Если ЕдШт = Неопределено Тогда
		ЕдШт = EnsureUnitSht();
	КонецЕсли;
	SetFieldIfExists(Эл, "ЕдиницаИзмерения", ЕдШт);
	SetFieldIfExists(Эл, "Единица", ЕдШт);
	SetFieldIfExists(Эл, "ЕдиницаИзмеренияХранения", ЕдШт);

	Эл.Записать();

	Возврат Эл.Ссылка;

Процедура EnsureOrderPaymentUiFields(Док) Экспорт
	// 1) Банковский счет (если реквизит есть и пуст)
	Попытка
		Тек = GetFieldValueSafe(Док, "БанковскийСчет");
		Если НЕ ЗначениеЗаполнено(Тек) Тогда
			// Сначала попробуем взять "основной" банковский счёт из организации
			Орг0 = GetFieldValueSafe(Док, "Организация");
			Сс = TryGetBankAccountFromOrg(Орг0);

			// В вашей УНФ тип реквизита = "Банковский счет" (не строка).
			// Подберём первый доступный объект по типу.
			Если НЕ ЗначениеЗаполнено(Сс) Тогда
				ТипИмя = GetRequisiteTypeName(Док, "БанковскийСчет");
				Сс = ResolveRefByTypeLabel(ТипИмя, "", Истина);
			КонецЕсли;
			Если НЕ ЗначениеЗаполнено(Сс) Тогда
				Сс = TryFindBankAccountForOrg(Орг0);
			КонецЕсли;
			Если НЕ ЗначениеЗаполнено(Сс) Тогда
				Сс = EnsureBankAccountFallback();
			КонецЕсли;
			Если ЗначениеЗаполнено(Сс) Тогда
				SetFieldIfExists(Док, "БанковскийСчет", Сс);
				// проверка: если не заполнилось (не тот тип) — сбросим
				Если НЕ ЗначениеЗаполнено(GetFieldValueSafe(Док, "БанковскийСчет")) Тогда
					// ничего
				КонецЕсли;
			КонецЕсли;
		КонецЕсли;
	Исключение
	КонецПопытки;

	// 2) Условия счета заказа / текст условий — на многих формах влияет на отображение "Оплатить"
	Попытка
		// Тип реквизита = "Дополнительные условия" (не строка) — выставляем через smart setter
		SetRequisiteByTextSmart(Док, "УсловияСчетаЗаказа", "Предоплата");
		Если НЕ ЗначениеЗаполнено(GetFieldValueSafe(Док, "УсловияСчетаЗаказа")) Тогда
			// fallback: первый элемент типа
			ТипИмя2 = GetRequisiteTypeName(Док, "УсловияСчетаЗаказа");
			Сс2 = ResolveRefByTypeLabel(ТипИмя2, "", Истина);
			Если НЕ ЗначениеЗаполнено(Сс2) Тогда

&НаСервере
Процедура EnsureOrderTotalsFromItems(Док, PreferredSum) Экспорт
	Если Док = Неопределено Тогда Возврат; КонецЕсли;
	Если PreferredSum = Неопределено Тогда PreferredSum = 0; КонецЕсли;

	// В разных базах строки могут лежать в разных табчастях.
	// Поэтому суммируем по нескольким кандидатам и берем максимум.
	Кандидаты = Новый Массив;
	Кандидаты.Добавить("Запасы");
	Кандидаты.Добавить("Товары");
	Кандидаты.Добавить("ТоварыУслуги");
	Кандидаты.Добавить("ТоварыИУслуги");
	Кандидаты.Добавить("ТоварыРаботыУслуги");

	СуммаПоСтрокам = 0;
	Для Каждого ИмяТЧ Из Кандидаты Цикл
		ТЧ = TryGetTablePart(Док, ИмяТЧ);
		Если ТЧ = Неопределено Тогда Продолжить; КонецЕсли;
		Если ТЧ.Количество() = 0 Тогда Продолжить; КонецЕсли;

		СумТЧ = 0;
		Перв = Истина;
		Для Каждого Стр Из ТЧ Цикл
			Кол = ToNumber(GetFieldValueSafe(Стр, "Количество"));
			Если Кол = 0 Тогда Кол = ToNumber(GetFieldValueSafe(Стр, "КоличествоУпаковок")); КонецЕсли;
			Цена = ToNumber(GetFieldValueSafe(Стр, "Цена"));
			Если Цена = 0 Тогда Цена = ToNumber(GetFieldValueSafe(Стр, "ЦенаПродажи")); КонецЕсли;
			Если Цена = 0 Тогда Цена = ToNumber(GetFieldValueSafe(Стр, "ЦенаЕдиницы")); КонецЕсли;

			Сум = ToNumber(GetFieldValueSafe(Стр, "Сумма"));
			Если Сум = 0 Тогда Сум = ToNumber(GetFieldValueSafe(Стр, "СуммаСтроки")); КонецЕсли;
			Если Сум = 0 Тогда Сум = ToNumber(GetFieldValueSafe(Стр, "СуммаВсего")); КонецЕсли;
			// В вашей УНФ итог документа считается по колонке "Всего"
			ВсегоСтр = ToNumber(GetFieldValueSafe(Стр, "Всего"));
			Если ВсегоСтр <> 0 Тогда
				Сум = ВсегоСтр;
			ИначеЕсли Сум <> 0 Тогда
				// если "Всего" не заполнено, попробуем заполнить
				Попытка SetFieldIfExists(Стр, "Всего", Сум); Исключение КонецПопытки;
			КонецЕсли;
			Если Сум = 0 Тогда
				Сум = Окр(Кол * Цена, 2);
				Если Сум <> 0 Тогда
					SetFieldIfExists(Стр, "Сумма", Сум);
					SetFieldIfExists(Стр, "СуммаСтроки", Сум);
					SetFieldIfExists(Стр, "СуммаВсего", Сум);
				КонецЕсли;
			КонецЕсли;

			Если (Сум = 0) И (PreferredSum > 0) И Перв Тогда
				Кол = 1; Цена = PreferredSum; Сум = PreferredSum;
				SetFieldIfExists(Стр, "Количество", 1);
				SetFieldIfExists(Стр, "КоличествоУпаковок", 1);
				SetFieldIfExists(Стр, "Цена", PreferredSum);
				SetFieldIfExists(Стр, "ЦенаПродажи", PreferredSum);
				SetFieldIfExists(Стр, "ЦенаЕдиницы", PreferredSum);
				SetFieldIfExists(Стр, "Сумма", PreferredSum);
				SetFieldIfExists(Стр, "СуммаСтроки", PreferredSum);
				SetFieldIfExists(Стр, "СуммаВсего", PreferredSum);
			КонецЕсли;

			СумТЧ = СумТЧ + Сум;
			Перв = Ложь;
		КонецЦикла;

		Если СумТЧ > СуммаПоСтрокам Тогда
			СуммаПоСтрокам = СумТЧ;
		КонецЕсли;
	КонецЦикла;

	Если (СуммаПоСтрокам = 0) И (PreferredSum > 0) Тогда
		СуммаПоСтрокам = PreferredSum;
	КонецЕсли;

	Если СуммаПоСтрокам > 0 Тогда
		SetFieldIfExists(Док, "СуммаДокумента", СуммаПоСтрокам);
		SetFieldIfExists(Док, "Сумма", СуммаПоСтрокам);
		SetFieldIfExists(Док, "Итого", СуммаПоСтрокам);
		SetFieldIfExists(Док, "СуммаВсего", СуммаПоСтрокам);
		SetFieldIfExists(Док, "СуммаКОплате", СуммаПоСтрокам);
		SetFieldIfExists(Док, "ИтогоКОплате", СуммаПоСтрокам);
		SetFieldIfExists(Док, "СуммаПродажи", СуммаПоСтрокам);

		// На некоторых конфигурациях итоговые реквизиты пересчитываются только явным вызовом.
		Попытка Док.ПересчитатьИтоги(); Исключение КонецПопытки;
		Попытка Док.РассчитатьИтоги(); Исключение КонецПопытки;
	КонецЕсли;

Процедура SetRequisiteByTextSmart(Док, ИмяРеквизита, Текст) Экспорт
	// 1) Попробуем присвоить строкой
	SetFieldIfExists(Док, ИмяРеквизита, Текст);

	// 2) Попробуем перечисление/справочник по типу реквизита
	ТипИмя = GetRequisiteTypeName(Док, ИмяРеквизита);
	Если ПустаяСтрока(ТипИмя) Тогда Возврат; КонецЕсли;

	Если Лев(ТипИмя, 15) = "Перечисление" Тогда
		Позиция = Найти(ТипИмя, ".");
		Если Позиция > 0 Тогда
			ИмяПеречисления = Сред(ТипИмя, Позиция + 1);
			ЗнПереч = EnumValueByText(ИмяПеречисления, Текст);
			Если ЗнПереч <> Неопределено Тогда
				SetFieldIfExists(Док, ИмяРеквизита, ЗнПереч);
			ИначеЕсли (ИмяРеквизита = "Состояние") ИЛИ (ИмяРеквизита = "Статус") Тогда
				ЗнПереч = EnumFirstValue(ИмяПеречисления);
				Если ЗнПереч <> Неопределено Тогда SetFieldIfExists(Док, ИмяРеквизита, ЗнПереч); КонецЕсли;
			КонецЕсли;
		КонецЕсли;
	ИначеЕсли Лев(ТипИмя, 11) = "Справочник" Тогда
		Позиция = Найти(ТипИмя, ".");
		Если Позиция > 0 Тогда
			ИмяСпр = Сред(ТипИмя, Позиция + 1);
			Сс = CatalogRefByName(ИмяСпр, Текст);
			Если НЕ ЗначениеЗаполнено(Сс) И ((ИмяРеквизита = "Состояние") ИЛИ (ИмяРеквизита = "Статус")) Тогда
				Сс = CatalogFirstRef(ИмяСпр);
			КонецЕсли;
			Если ЗначениеЗаполнено(Сс) Тогда SetFieldIfExists(Док, ИмяРеквизита, Сс); КонецЕсли;
		КонецЕсли;
	Иначе
		// В УНФ многие ссылочные типы отображаются как "Контрагент", "Валюта", "Ставка НДС" и т.п.
		// Пытаемся разрешить по синониму типа через Метаданные.
		AllowFirst = (ИмяРеквизита = "СостояниеЗаказа") ИЛИ (ИмяРеквизита = "ВидОперации") ИЛИ (ИмяРеквизита = "НалогообложениеНДС");
		Зн = ResolveRefByTypeLabel(ТипИмя, Текст, AllowFirst);
		Если Зн <> Неопределено Тогда
			SetFieldIfExists(Док, ИмяРеквизита, Зн);
		КонецЕсли;
	КонецЕсли;
КонецПроцедуры

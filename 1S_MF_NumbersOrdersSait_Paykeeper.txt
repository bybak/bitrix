Процедура ПриУстановкеНовогоНомера(Источник, СтандартнаяОбработка, Префикс) Экспорт

	Если Не ЭтоЗаказПокупателя(Источник) Тогда
		Возврат;
	КонецЕсли;

	НомерСайта = ПолучитьНомерЗаказаНаСайте(Источник.Ссылка);

	Если НомерСайта = "" Тогда
		Возврат;
	КонецЕсли;

	Источник.Номер = НомерСайта;
	СтандартнаяОбработка = Ложь;

Процедура ПриЗаписи(Источник, Отказ, РежимЗаписи, РежимПроведения) Экспорт

	Если Отказ Тогда
		Возврат;
	КонецЕсли;

	Если Не ЭтоЗаказПокупателя(Источник) Тогда
		Возврат;
	КонецЕсли;

	// Запись заказа из PayKeeper — не заходим снова в обработчики.
	Если MF_ЭтоСлужебнаяЗаписьЗаказа(Источник) Тогда
		Возврат;
	КонецЕсли;

	НомерСайта = ПолучитьНомерЗаказаНаСайте(Источник.Ссылка);

	Если НомерСайта <> "" И СокрЛП(Источник.Номер) <> НомерСайта Тогда
		Источник.Номер = НомерСайта;
	КонецЕсли;

	ИсправитьЦеныЗаказаССайта(Источник);

	// PayKeeper: после обмена, если способ оплаты PayKeeper и счёт ещё не создан.
	АвтоPaykeeperОбработатьЗаказ(Источник.Ссылка);

Функция ПолучитьНомерЗаказаНаСайте(ЗаказСсылка)

	Если Не ЗначениеЗаполнено(ЗаказСсылка) Тогда
		Возврат "";
	КонецЕсли;

	ИменаРегистров = Новый Массив;
	ИменаРегистров.Добавить("ЗаказыПокупателейССайта");
	ИменаРегистров.Добавить("ЗаказыКлиентовССайта");
	ИменаРегистров.Добавить("ЗаказыССайта");

	ПоляЗаказа = Новый Массив;
	ПоляЗаказа.Добавить("ЗаказПокупателя");
	ПоляЗаказа.Добавить("ЗаказКлиента");
	ПоляЗаказа.Добавить("Заказ");

	ПоляНомера.Добавить("НомерЗаказаНаСайте");
	ПоляНомера.Добавить("НомерЗаказаСайта");
	ПоляНомера.Добавить("НомерНаСайте");

	Для Каждого ИмяРегистра Из ИменаРегистров Цикл

		Если Метаданные.РегистрыСведений.Найти(ИмяРегистра) = Неопределено Тогда
			Продолжить;
		КонецЕсли;

		Для Каждого ПолеЗаказа Из ПоляЗаказа Цикл
			Для Каждого ПолеНомера Из ПоляНомера Цикл

				Запрос = Новый Запрос;
				Запрос.Текст =
					"ВЫБРАТЬ ПЕРВЫЕ 1
					|	Данные." + ПолеНомера + " КАК НомерСайта
					|ИЗ
					|	РегистрСведений." + ИмяРегистра + " КАК Данные
					|ГДЕ
					|	Данные." + ПолеЗаказа + " = &Заказ";

				Запрос.УстановитьПараметр("Заказ", ЗаказСсылка);

				Попытка
					Выборка = Запрос.Выполнить().Выбрать();

					Если Выборка.Следующий() Тогда
						НомерСайта = СокрЛП(Строка(Выборка.НомерСайта));

						Если НомерСайта <> "" Тогда
							Возврат НомерСайта;
						КонецЕсли;
					КонецЕсли;
				Исключение
				КонецПопытки;

			КонецЦикла;
		КонецЦикла;

	КонецЦикла;

	Возврат "";

Процедура УстановитьНомерЗаказаПоСайту(ЗаказСсылка, НомерСайта) Экспорт

	Если Не ЗначениеЗаполнено(ЗаказСсылка) Или НомерСайта = "" Тогда
		Возврат;
	КонецЕсли;

	ЗаказОбъект = ЗаказСсылка.ПолучитьОбъект();

	Если СокрЛП(Строка(ЗаказОбъект.Номер)) <> НомерСайта Тогда
		ЗаказОбъект.Номер = НомерСайта;
	КонецЕсли;

	ИсправитьЦеныЗаказаССайтаБезПроверки(ЗаказОбъект);

	Если ЗаказОбъект.Проведен Тогда
		ЗаказОбъект.Записать(РежимЗаписиДокумента.Проведение);
	Иначе
		ЗаказОбъект.Записать(РежимЗаписиДокумента.Запись);
	КонецЕсли;

	// После привязки к сайту — PayKeeper (идемпотентно).
	АвтоPaykeeperОбработатьЗаказ(ЗаказСсылка);

	Результат = Новый Массив;

Функция ВырезатьДокументXML(ТекстXML, НомерСайта)

	ПозицияНомера = Найти(ТекстXML, "<Номер>" + НомерСайта + "</Номер>");

	Если ПозицияНомера = 0 Тогда
		Возврат "";
	КонецЕсли;

	НачалоДокумента = 0;
	ПозицияПоиска = ПозицияНомера;

	Пока ПозицияПоиска > 0 Цикл
		Фрагмент = Лев(ТекстXML, ПозицияПоиска);
		Позиция = НайтиСКонца(Фрагмент, "<Документ");

		Если Позиция > 0 Тогда
			НачалоДокумента = Позиция;
			Прервать;
		КонецЕсли;

		Прервать;
	КонецЦикла;

	Если НачалоДокумента = 0 Тогда
		Возврат "";
	КонецЕсли;

	КонецОтносительно = Найти(Сред(ТекстXML, ПозицияНомера), "</Документ>");

	Если КонецОтносительно = 0 Тогда
		Возврат "";
	КонецЕсли;

	КонецДокумента = ПозицияНомера + КонецОтносительно + СтрДлина("</Документ>") - 2;

	Возврат Сред(ТекстXML, НачалоДокумента, КонецДокумента - НачалоДокумента + 1);

Функция ПолучитьЦеныИзДокументаXML(ДокументXML)

	Результат = Новый Массив;
	Позиция = 1;

	Пока Истина Цикл

		НачалоТовараОтн = Найти(Сред(ДокументXML, Позиция), "<Товар");

		Если НачалоТовараОтн = 0 Тогда
			Прервать;
		КонецЕсли;

		НачалоТовара = Позиция + НачалоТовараОтн - 1;
		КонецТовараОтн = Найти(Сред(ДокументXML, НачалоТовара), "</Товар>");

		Если КонецТовараОтн = 0 Тогда
			Прервать;
		КонецЕсли;

		КонецТовара = НачалоТовара + КонецТовараОтн + СтрДлина("</Товар>") - 2;
		ТоварXML = Сред(ДокументXML, НачалоТовара, КонецТовара - НачалоТовара + 1);

		Цена = ПолучитьЗначениеРеквизитаXML(ТоварXML, "Цена заказа сайта MF");

		Если Цена <= 0 Тогда
			Цена = ЧислоИзXML(ПолучитьЗначениеТегаXML(ТоварXML, "ЦенаЗаЕдиницу"));
		КонецЕсли;

		Если Цена > 0 Тогда
			Результат.Добавить(Цена);
		КонецЕсли;

		Позиция = КонецТовара + 1;

	КонецЦикла;

	Возврат Результат;

Функция ПолучитьЗначениеРеквизитаXML(ТоварXML, ИмяРеквизита)

	ПозицияИмени = Найти(ТоварXML, "<Наименование>" + ИмяРеквизита + "</Наименование>");

	Если ПозицияИмени = 0 Тогда
		Возврат 0;

	PaykeeperЗаписатьРеквизит(ЗаказОбъект, "Доп_НомерЗаказа", НомерЗаказа);

	PaykeeperСохранитьHtmlСчета(ЗаказОбъект, Данные.invoice);

	MF_ПометитьСлужебнуюЗаписьЗаказа(ЗаказОбъект);
	Если ЗаказОбъект.Проведен Тогда
		ЗаказОбъект.Записать(РежимЗаписиДокумента.Проведение);
	Иначе
		ЗаказОбъект.Записать(РежимЗаписиДокумента.Запись);
	КонецЕсли;

Процедура PaykeeperОтправитьПисьмо(ЗаказОбъект, Тип, Почта)

	Если Строка(Тип) <> "1" Тогда
		Возврат;
	КонецЕсли;

	Сервер = СокрЛП(ЗаказОбъект.Организация.Paykeeper_Сервер);
	Логин = СокрЛП(ЗаказОбъект.Организация.Paykeeper_Логин);
	Пароль = СокрЛП(ЗаказОбъект.Организация.Paykeeper_Пароль);

	ssl1 = Новый ЗащищенноеСоединениеOpenSSL(
		Новый СертификатКлиентаWindows(),
		Новый СертификатыУдостоверяющихЦентровWindows());

	Соединение = Новый HTTPСоединение(Сервер,, Логин, Пароль,,, ssl1);

	Заголовки = Новый Соответствие;
	Заголовки.Вставить("Content-type", "application/x-www-form-urlencoded");

	Запрос = Новый HTTPЗапрос("/info/settings/token/", Заголовки);
	Ответ = Соединение.Получить(Запрос);

	Если Ответ.КодСостояния <> 200 Тогда
		ВызватьИсключение "PayKeeper token (mail) HTTP " + Строка(Ответ.КодСостояния);
	КонецЕсли;

	ЧтениеJSON = Новый ЧтениеJSON;
	ЧтениеJSON.УстановитьСтроку(Ответ.ПолучитьТелоКакСтроку());
	ДанныеТокен = ПрочитатьJSON(ЧтениеJSON);
	ЧтениеJSON.Закрыть();

	OrderId = PaykeeperПрочитатьРеквизит(ЗаказОбъект, "Доп_OrderId");
	Текст = "token=" + ДанныеТокен.token + "&id=" + OrderId;

	Запрос = Новый HTTPЗапрос("/change/invoice/send/", Заголовки);
	Запрос.УстановитьТелоИзСтроки(Текст);
	Ответ = Соединение.ОтправитьДляОбработки(Запрос);

	Если Ответ.КодСостояния <> 200 Тогда
		ВызватьИсключение "PayKeeper invoice send HTTP " + Строка(Ответ.КодСостояния);
	КонецЕсли;

			Дан.tax = osnСоотв.Получить(ЗаказОбъект.СтавкаНДСДоставки);
			Дан.payment_type = "prepay";
			Дан.item_type = "service";

Процедура ПриУстановкеНовогоНомера(Источник, СтандартнаяОбработка, Префикс) Экспорт

	Если Не ЭтоЗаказПокупателя(Источник) Тогда
		Возврат;
	КонецЕсли;

	НомерСайта = MF_ПолучитьНомерСайтаДляОбъекта(Источник);

	Если НомерСайта = "" Тогда
		НомерСайта = MF_ВзятьНомерСайтаИзКэшаСеанса();
	КонецЕсли;

	Если НомерСайта = "" Тогда
		Возврат;
	КонецЕсли;

	Источник.Номер = НомерСайта;
	СтандартнаяОбработка = Ложь;

КонецПроцедуры


Процедура ПриЗаписи(Источник, Отказ, РежимЗаписи, РежимПроведения) Экспорт

	Если Отказ Тогда
		Возврат;
	КонецЕсли;

	Если Не ЭтоЗаказПокупателя(Источник) Тогда
		Возврат;
	КонецЕсли;

	Если MF_ЭтоСлужебнаяЗаписьЗаказа(Источник) Тогда
		Возврат;
	КонецЕсли;

	MF_УстановитьНомерСайтаВОбъекте(Источник);
	MF_ИсправитьНомерCAПриЗаписи(Источник);
	ИсправитьЦеныЗаказаССайта(Источник);

	Если ЭтоЗаказССайта(Источник.Ссылка) Тогда
		MF_УстановитьДатуОтгрузкиЗаказаССайта(Источник);
	КонецЕсли;

Функция ПолучитьНомерЗаказаНаСайте(ЗаказСсылка)

	Если Не ЗначениеЗаполнено(ЗаказСсылка) Тогда
		Возврат "";
	КонецЕсли;

	ИменаРегистров = Новый Массив;
	ИменаРегистров.Добавить("ЗаказыПокупателейССайта");
	ИменаРегистров.Добавить("ЗаказыКлиентовССайта");
	ИменаРегистров.Добавить("ЗаказыССайта");

	ПоляЗаказа = Новый Массив;
	ПоляЗаказа.Добавить("ЗаказПокупателя");
	ПоляЗаказа.Добавить("ЗаказКлиента");
	ПоляЗаказа.Добавить("Заказ");

	ПоляНомера = Новый Массив;
	ПоляНомера.Добавить("НомерЗаказаНаСайте");
	ПоляНомера.Добавить("НомерЗаказаСайта");
	ПоляНомера.Добавить("НомерНаСайте");

	Для Каждого ИмяРегистра Из ИменаРегистров Цикл

		Если Метаданные.РегистрыСведений.Найти(ИмяРегистра) = Неопределено Тогда
			Продолжить;
		КонецЕсли;

		Для Каждого ПолеЗаказа Из ПоляЗаказа Цикл
			Для Каждого ПолеНомера Из ПоляНомера Цикл

				Запрос = Новый Запрос;
				Запрос.Текст =
					"ВЫБРАТЬ ПЕРВЫЕ 1
					|	Данные." + ПолеНомера + " КАК НомерСайта
					|ИЗ
					|	РегистрСведений." + ИмяРегистра + " КАК Данные
					|ГДЕ
					|	Данные." + ПолеЗаказа + " = &Заказ";

				Запрос.УстановитьПараметр("Заказ", ЗаказСсылка);

				Попытка
					Выборка = Запрос.Выполнить().Выбрать();

					Если Выборка.Следующий() Тогда
						НомерСайта = СокрЛП(Строка(Выборка.НомерСайта));

						Если НомерСайта <> "" Тогда
							Возврат НомерСайта;
						КонецЕсли;
					КонецЕсли;
				Исключение
				КонецПопытки;

			КонецЦикла;
		КонецЦикла;

	КонецЦикла;

	Возврат "";

Функция MF_ПолучитьНомерСайтаДляОбъекта(ЗаказОбъект)

	Если ЗаказОбъект = Неопределено Тогда
		Возврат "";
	КонецЕсли;

	НомерСайта = ПолучитьНомерЗаказаНаСайте(ЗаказОбъект.Ссылка);
	Если НомерСайта <> "" Тогда
		Возврат НомерСайта;
	КонецЕсли;

	НомерСайта = MF_ПолучитьНомерСайтаИзДопСвойствОбмена(ЗаказОбъект);
	Если НомерСайта <> "" Тогда
		Возврат НомерСайта;
	КонецЕсли;

	ИменаРеквизитов = Новый Массив;
	ИменаРеквизитов.Добавить("Номер заказа сайта");
	ИменаРеквизитов.Добавить("Номер заказа на сайте");
	ИменаРеквизитов.Добавить("Номер на сайте");
	ИменаРеквизитов.Добавить("Номер заказа MF");
	ИменаРеквизитов.Добавить("Идентификатор заказа");

	Для Каждого ИмяРеквизита Из ИменаРеквизитов Цикл
		Значение = ПолучитьДопСведениеЗаказаСтрокой(ЗаказОбъект.Ссылка, ИмяРеквизита);
		Если Значение <> "" Тогда
			Возврат Значение;
		КонецЕсли;
	КонецЦикла;

	Возврат "";

Функция MF_ЭтоНомерОбменаССайтом(Номер)

	Номер = СокрЛП(Строка(Номер));
	Если Номер = "" Тогда
		Возврат Ложь;
	КонецЕсли;

	// Номер сайта: 36-372 (начинается с цифры).
	ПервыйСимвол = Лев(Номер, 1);
	Если ПервыйСимвол >= "0" И ПервыйСимвол <= "9" Тогда
		Возврат Ложь;
	КонецЕсли;

	// Номер обмена: CA-*, НФНФ-* (буквенный префикс).
	Если СтрНайти(Номер, "-") = 0 Тогда
		Возврат Ложь;
	КонецЕсли;

	Возврат Истина;

Процедура MF_ПометитьНаУдалениеДубльОбмена(ЦелевойЗаказ, НомерСайта)

	Если Не ЗначениеЗаполнено(ЦелевойЗаказ) Или НомерСайта = "" Тогда
		Возврат;
	КонецЕсли;

	ЦелевойОбъект = ЦелевойЗаказ.ПолучитьОбъект();
	Если ЦелевойОбъект = Неопределено Тогда
		Возврат;
	КонецЕсли;

	Контрагент = ЦелевойОбъект.Контрагент;
	ДатаЗаказа = ЦелевойОбъект.Дата;
	Если Не ЗначениеЗаполнено(Контрагент) Или Не ЗначениеЗаполнено(ДатаЗаказа) Тогда
		Возврат;
	КонецЕсли;

	НачалоДня = НачалоДня(ДатаЗаказа);
	КонецДня = КонецДня(ДатаЗаказа);

	// Удаляем CA-* только если уже есть «правильный» заказ с номером сайта.
	ЗапросПроверка = Новый Запрос;
	ЗапросПроверка.Текст =
		"ВЫБРАТЬ ПЕРВЫЕ 1
		|	Заказ.Ссылка КАК Ссылка
		|ИЗ
		|	Документ.ЗаказПокупателя КАК Заказ
		|ГДЕ
		|	Заказ.Контрагент = &Контрагент
		|	И Заказ.Дата МЕЖДУ &НачалоДня И &КонецДня
		|	И Заказ.Номер = &НомерСайта
		|	И НЕ Заказ.ПометкаУдаления";

	ЗапросПроверка.УстановитьПараметр("Контрагент", Контрагент);
	ЗапросПроверка.УстановитьПараметр("НачалоДня", НачалоДня);
	ЗапросПроверка.УстановитьПараметр("КонецДня", КонецДня);
	ЗапросПроверка.УстановитьПараметр("НомерСайта", НомерСайта);

	Попытка
		Если ЗапросПроверка.Выполнить().Пустой() Тогда
			Возврат;
		КонецЕсли;
	Исключение
		Возврат;
	КонецПопытки;

	MF_УдалитьДублиCAЗаДень(Контрагент, НачалоДня, КонецДня, НомерСайта);

// Сразу при записи: второй/третий дубль 36-XXX помечаем на удаление.
Процедура MF_СразуУбратьДублиПриЗаписи(ЗаказСсылка, НомерСайта)

	НомерСайта = СокрЛП(Строка(НомерСайта));
	Если НомерСайта = "" Или Не ЗначениеЗаполнено(ЗаказСсылка) Тогда
		Возврат;
	КонецЕсли;

	Канонический = MF_НайтиЛучшийЗаказПоНомеруСайта(НомерСайта);
	Если Не ЗначениеЗаполнено(Канонический) Тогда
		Возврат;
	КонецЕсли;

	Если Канонический <> ЗаказСсылка Тогда
		АвтоPaykeeperЛог(ЗаказСсылка, "дубль при записи — удаляем, оставляем " + Строка(Канонический));
		MF_ПометитьДокументНаУдаление(ЗаказСсылка);
		Возврат;
	КонецЕсли;

	MF_УбратьДублиНомераСайта(Канонический, НомерСайта);

// Дополнить таблицу обмена заказами, найденными по печатному номеру (36-373), а не только по регистру.
Процедура MF_ДополнитьРанееЗагруженныеЗаказы(МассивНомеров, РанееЗагруженныеЗаказы) Экспорт

	Если МассивНомеров = Неопределено Или РанееЗагруженныеЗаказы = Неопределено Тогда
		Возврат;
	КонецЕсли;

	Для Каждого НомерСайта Из МассивНомеров Цикл

		НомерСайта = СокрЛП(Строка(НомерСайта));
		Если НомерСайта = "" Тогда
			Продолжить;
		КонецЕсли;

		Если РанееЗагруженныеЗаказы.Найти(НомерСайта, "НомерЗаказаНаСайте") <> Неопределено Тогда
			Продолжить;
		КонецЕсли;

		ЗаказСсылка = MF_НайтиЛучшийЗаказПоНомеруСайта(НомерСайта);
		Если Не ЗначениеЗаполнено(ЗаказСсылка) Тогда
			ЗаказСсылка = MF_НайтиЗаказПоХвостуНомераСайта(НомерСайта);
		КонецЕсли;
		Если Не ЗначениеЗаполнено(ЗаказСсылка) Тогда
			Продолжить;
		КонецЕсли;

		НоваяСтрока = РанееЗагруженныеЗаказы.Добавить();
		НоваяСтрока.ЗаказПокупателя = ЗаказСсылка;
		НоваяСтрока.НомерЗаказаНаСайте = НомерСайта;
		НоваяСтрока.ЕстьСсылки = Ложь;

	КонецЦикла;

// Список строк через "|" — MF_МассивИзСтрок("a","b") в 1С недопустим (только числа-размеры).
Функция MF_МассивИзСтрок(СтрокаСписка)

	Результат = Новый Массив;
	СтрокаСписка = СокрЛП(Строка(СтрокаСписка));

	Если СтрокаСписка = "" Тогда
		Возврат Результат;
	КонецЕсли;

	Попытка
		Части = СтрРазделить(СтрокаСписка, "|", Ложь);
		Для Каждого Часть Из Части Цикл
			Текст = СокрЛП(Строка(Часть));
			Если Текст <> "" Тогда
				Результат.Добавить(Текст);
			КонецЕсли;
		КонецЦикла;
		Возврат Результат;
	Исключение
	КонецПопытки;

	Остаток = СтрокаСписка;
	Пока Остаток <> "" Цикл
		Позиция = СтрНайти(Остаток, "|");
		Если Позиция = 0 Тогда
			Текст = СокрЛП(Остаток);
			Если Текст <> "" Тогда
				Результат.Добавить(Текст);
			КонецЕсли;
			Прервать;
		КонецЕсли;
		Текст = СокрЛП(Лев(Остаток, Позиция - 1));
		Если Текст <> "" Тогда
			Результат.Добавить(Текст);
		КонецЕсли;
		Остаток = СокрЛП(Сред(Остаток, Позиция + 1));
	КонецЦикла;

	Возврат Результат;

// Диагностика: в консоли запросов — MF_NumbersOrdersSaitServer.MF_ДиагностикаPaykeeperДляЗаказа(СсылкаНаЗаказ);

	Если Не ЗначениеЗаполнено(ЗаказСсылка) Тогда

			НастройкиPK = MF_ПолучитьНастройкиPaykeeperОрганизации(Объект.Организация);

		Если ОбъектОтмена <> Неопределено И MF_ЗаказУжеОтмененВ1С(ОбъектОтмена) Тогда

	Исключение
	КонецПопытки;

	Если Не MF_ЗаказПроведен(ЗаказСсылка) Тогда
		НомерДляПроведения = "";
		Попытка
			ОбъектПроведение = ЗаказСсылка.ПолучитьОбъект();
			Если ОбъектПроведение <> Неопределено Тогда
				НомерДляПроведения = СокрЛП(Строка(ОбъектПроведение.Номер));

		Если НомерДляПроведения <> "" Тогда
			АвтоPaykeeperЛог(ЗаказСсылка, "догоняем проведение перед PayKeeper");

		КонецЕсли;

		АвтоPaykeeperЛог(ЗаказСсылка, "пропуск: письмо PayKeeper уже отправлено в этой сессии");

		АвтоPaykeeperЛог(ЗаказСсылка, "пропуск: PayKeeper уже обрабатывается в этой сессии");
		Возврат;

	MF_PaykeeperУстановитьБлокировку(ЗаказСсылка, Истина);

				АвтоPaykeeperЛог(ЗаказСсылка, "пропуск: заказ не с сайта");
				Возврат;
			КонецЕсли;
			АвтоPaykeeperЛог(ЗаказСсылка, "заказ 36-* — считаем заказом с сайта");
		КонецЕсли;

		ЭтоPaykeeper = ЭтоОплатаPaykeeper(ЗаказСсылка);
		Если Не ЭтоPaykeeper Тогда
			НомерДляЛога = "";
			Попытка
				ОбъектЛог = ЗаказСсылка.ПолучитьОбъект();
				Если ОбъектЛог <> Неопределено Тогда
					НомерДляЛога = СокрЛП(Строка(ОбъектЛог.Номер));
				КонецЕсли;

				ТекстКэша = " cache[метод=" + ДанныеКэша.МетодОплаты + ", MF=" + ДанныеКэша.MFКод + ", id=" + ДанныеКэша.МетодОплатыИД + "]";
			КонецЕсли;
			АвтоPaykeeperЛог(ЗаказСсылка, "пропуск: способ оплаты не PayKeeper (метод="
				+ ПолучитьДопСведениеЗаказаСтрокой(ЗаказСсылка, "Метод оплаты")
				+ ", MF=" + ПолучитьДопСведениеЗаказаСтрокой(ЗаказСсылка, "MF_PAY_SYSTEM")
				+ ", id=" + ПолучитьДопСведениеЗаказаСтрокой(ЗаказСсылка, "Метод оплаты ИД")
				+ ", MF_PAYKEEPER=" + ПолучитьДопСведениеЗаказаСтрокой(ЗаказСсылка, "MF_PAYKEEPER")

			Возврат;

		Если ЗаказОбъект = Неопределено Тогда
			АвтоPaykeeperЛог(ЗаказСсылка, "пропуск: не удалось получить объект заказа");
			Возврат;

		Если PaykeeperСчетУжеСоздан(ЗаказОбъект) Тогда
			АвтоPaykeeperЛог(ЗаказСсылка, "пропуск: счёт PayKeeper уже создан");
			Возврат;
		КонецЕсли;

			MF_ОбновитьКонтактыКонтрагентаПоЗаказу(ЗаказОбъект);

		КонецЕсли;

Процедура PaykeeperСоздатьСчетИОтправитьПисьмо(ЗаказОбъект, ОтветПочты)

	Сервер = НастройкиPK.Сервер;
	Логин = НастройкиPK.Логин;
	Пароль = НастройкиPK.Пароль;

	ssl1 = Новый ЗащищенноеСоединениеOpenSSL(
		Новый СертификатКлиентаWindows(),
		Новый СертификатыУдостоверяющихЦентровWindows());

	Заголовки = Новый Соответствие;

	Запрос = Новый HTTPЗапрос("/info/settings/token/", Заголовки);
	Ответ = Соединение.Получить(Запрос);

	Если Ответ.КодСостояния <> 200 Тогда
		ВызватьИсключение "PayKeeper token HTTP " + Строка(Ответ.КодСостояния);
	КонецЕсли;

	ЧтениеJSON.УстановитьСтроку(Ответ.ПолучитьТелоКакСтроку());
	ДанныеТокен = ПрочитатьJSON(ЧтениеJSON);
	ЧтениеJSON.Закрыть();

	НомерЗаказа = Лев(Строка(ЗаказОбъект.Ссылка.УникальныйИдентификатор()), 28);
	СтрокаСуммаДокумента = PaykeeperНормализоватьЧислоСтрокой(ЗаказОбъект.СуммаДокумента);
	МассивТоваров = PaykeeperСобратьКорзину(ЗаказОбъект);

	ЗаписьJSON = Новый ЗаписьJSON;
	ЗаписьJSON.УстановитьСтроку();
	ЗаписатьJSON(ЗаписьJSON, МассивТоваров);
	РезультатJSON = ЗаписьJSON.Закрыть();

	Текст = "token=" + ДанныеТокен.token
		+ "&clientid=" + Строка(ЗаказОбъект.Контрагент)
		+ "&pay_amount=" + СтрокаСуммаДокумента
		+ "&orderid=" + НомерЗаказа
		+ "&client_email=" + ОтветПочты.Почта
		+ "&service_name=;PKC|" + РезультатJSON + "|";

	Запрос = Новый HTTPЗапрос("/change/invoice/preview/", Заголовки);
	Запрос.УстановитьТелоИзСтроки(Текст);
	Ответ = Соединение.ОтправитьДляОбработки(Запрос);

	Если Ответ.КодСостояния <> 200 Тогда
		ВызватьИсключение "PayKeeper invoice preview HTTP " + Строка(Ответ.КодСостояния)
			+ " " + Лев(Ответ.ПолучитьТелоКакСтроку(), 300);
	КонецЕсли;

	ЧтениеJSON = Новый ЧтениеJSON;
	ЧтениеJSON.УстановитьСтроку(Ответ.ПолучитьТелоКакСтроку());
	Данные = ПрочитатьJSON(ЧтениеJSON);
	ЧтениеJSON.Закрыть();

	InvoiceId = "";
	InvoiceURL = "";
	InvoiceHTML = "";

	Попытка
		InvoiceId = СокрЛП(Строка(Данные.invoice_id));
	Исключение
	КонецПопытки;
	Если InvoiceId = "" Тогда
		Попытка
			InvoiceId = СокрЛП(Строка(Данные.id));
		Исключение
		КонецПопытки;
	КонецЕсли;

	Попытка
		InvoiceURL = СокрЛП(Строка(Данные.invoice_url));
	Исключение
	КонецПопытки;
	Попытка
		InvoiceHTML = Строка(Данные.invoice);
	Исключение
	КонецПопытки;

	Если InvoiceId = "" Или InvoiceURL = "" Тогда
		ВызватьИсключение "PayKeeper: пустой ответ invoice_id/invoice_url";
	КонецЕсли;

	PaykeeperЗаписатьРеквизит(ЗаказОбъект, "Доп_OrderId", InvoiceId);
	PaykeeperЗаписатьРеквизит(ЗаказОбъект, "Доп_OrderURL", InvoiceURL);
	PaykeeperЗаписатьРеквизит(ЗаказОбъект, "Доп_QR", InvoiceURL + "?pstype=sbp_default");
	PaykeeperЗаписатьРеквизит(ЗаказОбъект, "Доп_НомерЗаказа", НомерЗаказа);

	PaykeeperСохранитьHtmlСчета(ЗаказОбъект, InvoiceHTML);

	MF_ПометитьСлужебнуюЗаписьЗаказа(ЗаказОбъект);
	Если ЗаказОбъект.Проведен Тогда
		ЗаказОбъект.Записать(РежимЗаписиДокумента.Проведение);
	Иначе
		ЗаказОбъект.Записать(РежимЗаписиДокумента.Запись);
	КонецЕсли;

	Если Не MF_PaykeeperПисьмоУжеОтправлено(ЗаказОбъект.Ссылка) Тогда
		PaykeeperОтправитьПисьмо(ЗаказОбъект, ОтветПочты.Тип, ОтветПочты.Почта);
	КонецЕсли;

		Пока Выборка.Следующий() Цикл

	Если Не ЗначениеЗаполнено(КонтрагентСсылка) Тогда
		Возврат;
	КонецЕсли;

	Кэш = MF_ПолучитьКонтактыИзКэшаКонтрагента(КонтрагентСсылка);
	Если Кэш <> Неопределено Тогда
		Если СокрЛП(Телефон) = "" Тогда
			Телефон = СокрЛП(Строка(Кэш.Телефон));
		КонецЕсли;
		Если СокрЛП(Email) = "" Или Найти(Email, "@") = 0 Тогда
			Email = СокрЛП(Строка(Кэш.Email));
		КонецЕсли;
	КонецЕсли;

	ЗаказСсылка = MF_НайтиПоследнийЗаказКонтрагента(КонтрагентСсылка);
	Если Не ЗначениеЗаполнено(ЗаказСсылка) Тогда
		Возврат;
	КонецЕсли;

	ЗаказОбъект = Неопределено;
	Попытка
		ЗаказОбъект = ЗаказСсылка.ПолучитьОбъект();
	Исключение
		Возврат;
	КонецПопытки;

	Если ЗаказОбъект = Неопределено Тогда
		Возврат;
	КонецЕсли;

	НомерСайта = "";
	Попытка
		НомерСайта = СокрЛП(Строка(ЗаказОбъект.Номер));
	Исключение
	КонецПопытки;

	MF_ОбновитьКэшEmailИзДопСведений(ЗаказСсылка, НомерСайта);
	MF_ОбновитьКэшТелефонаИзДопСведений(ЗаказСсылка, НомерСайта);

	Если СокрЛП(Телефон) = "" Тогда
		Телефон = ПолучитьТелефонЗаказа(ЗаказОбъект);
	КонецЕсли;

	Если СокрЛП(Email) = "" Или Найти(Email, "@") = 0 Тогда
		Email = ПолучитьEmailЗаказа(ЗаказОбъект);
	КонецЕсли;

Функция MF_НайтиПоследнийЗаказКонтрагента(КонтрагентСсылка)

	Если Не ЗначениеЗаполнено(КонтрагентСсылка) Тогда
		Возврат Неопределено;
	КонецЕсли;

	ИменаДокументов = Новый Массив;
	ИменаДокументов.Добавить("ЗаказПокупателя");
	ИменаДокументов.Добавить("ЗаказКлиента");

	Для Каждого ИмяДокумента Из ИменаДокументов Цикл
		Попытка
			Если Метаданные.Документы.Найти(ИмяДокумента) = Неопределено Тогда
				Продолжить;
			КонецЕсли;
			Запрос = Новый Запрос;
			Запрос.Текст =
			"ВЫБРАТЬ ПЕРВЫЕ 1
			|	Заказ.Ссылка КАК Ссылка
			|ИЗ
			|	Документ." + ИмяДокумента + " КАК Заказ
			|ГДЕ
			|	Заказ.Контрагент = &Контрагент
			|	И НЕ Заказ.ПометкаУдаления
			|УПОРЯДОЧИТЬ ПО
			|	Заказ.Дата УБЫВ";
			Запрос.УстановитьПараметр("Контрагент", КонтрагентСсылка);
			Выборка = Запрос.Выполнить().Выбрать();
			Если Выборка.Следующий() Тогда
				Возврат Выборка.Ссылка;
			КонецЕсли;
		Исключение
		КонецПопытки;
	КонецЦикла;

	Возврат Неопределено;

Процедура MF_СохранитьКонтактыВКэшПоЗаказу(ЗаказОбъект, Телефон, Email)

	Если ЗаказОбъект = Неопределено Или Не ЗначениеЗаполнено(ЗаказОбъект.Контрагент) Тогда
		Возврат;
	КонецЕсли;

	Телефон = СокрЛП(Строка(Телефон));
	Email = СокрЛП(Строка(Email));
	Если Телефон = "" И (Email = "" Или Найти(Email, "@") = 0) Тогда
		Возврат;
	КонецЕсли;

	Данные = Новый Структура("Телефон, Email", Телефон, Email);
	КлючКэша = MF_КлючКэшаКонтактовКонтрагента(ЗаказОбъект.Контрагент);
	Попытка
		ПараметрыСеанса.Вставить(КлючКэша, Данные);
	Исключение
	КонецПопытки;

Функция ParseIsoDateSafe(S) Экспорт
	S = СокрЛП(Строка(S));
	Если ПустаяСтрока(S) Тогда Возврат Неопределено; КонецЕсли;
	П = Найти(S, "+");
	Если П > 0 Тогда S = Лев(S, П - 1); КонецЕсли;
	П = Найти(S, "Z");
	Если П > 0 Тогда S = Лев(S, П - 1); КонецЕсли;
	S = СтрЗаменить(S, "T", " ");
	Если СтрДлина(S) >= 10 И Найти(S, "-") > 0 Тогда
		Попытка
			Год = Число(Лев(S, 4));
			Мес = Число(Сред(S, 6, 2));
			Дн = Число(Сред(S, 9, 2));
			Ч = 0; Мн = 0; Сек = 0;
			Если СтрДлина(S) >= 19 Тогда
				Ч = Число(Сред(S, 12, 2));
				Мн = Число(Сред(S, 15, 2));
				Сек = Число(Сред(S, 18, 2));
			КонецЕсли;
			Возврат Дата(Год, Мес, Дн, Ч, Мн, Сек);
		Исключение
		КонецПопытки;
	КонецЕсли;
	Попытка
		Возврат Дата(S);
	Исключение
		Возврат Неопределено;
	КонецПопытки;
КонецФункции

Функция SupplierOrderLooksInProgress(ДокОбъект, КлючСостояния) Экспорт
	// Ключ: подстрока в нижнем регистре, напр. "в работе" / "вработе" (см. заголовок X-MF-State-Key)
	Ключ = СокрЛП(НРег(Строка(КлючСостояния)));
	Если ПустаяСтрока(Ключ) Тогда Ключ = "в работе"; КонецЕсли;
	Ключ2 = СтрЗаменить(Ключ, " ", "");

	Сыр = SupplierOrderStateRaw(ДокОбъект);
	Т0 = СокрЛП(Сыр.Текст);
	Т = НРег(Т0);
	Тс = СтрЗаменить(Т, " ", "");

	Если Найти(Т, Ключ) > 0 Тогда Возврат Истина; КонецЕсли;
	Если Найти(Тс, Ключ2) > 0 Тогда Возврат Истина; КонецЕсли;
	// Часто в УНФ: "Заказ выполняется" и т.п. — мягкое «работ»
	Если Найти(Т, "работ") > 0 Тогда Возврат Истина; КонецЕсли;
	Возврат Ложь;

Функция SupplierOrders(Запрос) Экспорт
	// Сигнатура для HTTP: Функция(HTTPСервисЗапрос) -> HTTPСервисОтвет

	Попытка
		// --- token ---
		Ож = СокрЛП(MfSupplierOrdersExpectedToken());
		Если НЕ ПустаяСтрока(Ож) Тогда
			Ток = СокрЛП(HeaderValue(Запрос, "X-MF-Token"));
			Если Ток <> Ож Тогда
				Возврат ОтветJSON(401, "{""ok"":false,""error"":""unauthorized""}");
			КонецЕсли;
		КонецЕсли;

		// --- debug version ---
		Дб = СокрЛП(ВРег(HeaderValue(Запрос, "X-MF-Debug")));
		Если (Дб = "VER") Или (Дб = "VERSION") Тогда
			С0 = Новый Структура("ok, script_version", Истина, MfSupplierOrdersScriptVersion());
			Возврат ОтветJSON(200, ЗаписатьСтруктуруВJSON(С0));
		КонецЕсли;

		// --- период: по умолчанию вся история; X-MF-From / X-MF-To сузить ---
		FromDef = Дата(2000, 1, 1);
		ToDef = КонецДня(ТекущаяДата());
		FromDate = ParseDateHeaderSafe(Запрос, "X-MF-From", FromDef);
		ToDate = ParseDateHeaderSafe(Запрос, "X-MF-To", ToDef);
		Попытка ToDate = КонецДня(ToDate); Исключение КонецПопытки;
		Попытка FromDate = НачалоДня(FromDate); Исключение КонецПопытки;

		// --- X-MF-Offset: пропуск N подходящих; X-MF-Limit: 0 (по умолчанию) = без лимита, >0 = не больше N ---
		Offset = ParseIntSafe(HeaderValue(Запрос, "X-MF-Offset"), 0);
		Limit = ParseIntSafe(HeaderValue(Запрос, "X-MF-Limit"), 0);
		Если Limit < 0 Тогда Limit = 0; КонецЕсли;
		Если Offset < 0 Тогда Offset = 0; КонецЕсли;

		КлючСостояния = СокрЛП(HeaderValue(Запрос, "X-MF-State-Key"));
		Если ПустаяСтрока(КлючСостояния) Тогда КлючСостояния = "в работе"; КонецЕсли;

		// --- менеджер (для fallback) + запрос «новейшие по дате» (основной путь) ---
		Мен = TryGetDocManagerByName("ЗаказПоставщику");
		Если Мен = Неопределено Тогда
			Мен = TryGetDocManagerByName("ЗаказПоставщика");
		КонецЕсли;
		Если Мен = Неопределено Тогда
			Возврат ОтветJSON(500, "{""ok"":false,""script_version"":""" + JsonEscape(MfSupplierOrdersScriptVersion()) + """,""error"":""doc_manager_not_found""}");
		КонецЕсли;

		// 0 = в запросе без «ПЕРВЫХ» (все проведённые в период), дальше отсекаем «не в работе» в цикле
		СписокСсылок = НовейшиеСсылкиПроведённыхПоПериоду(FromDate, ToDate, 0);
		РежимВыборки = "query_newest";
		Выборка = Неопределено;
		Если СписокСсылок = Неопределено Тогда
			РежимВыборки = "manager_select";
			Попытка
				Выборка = Мен.Выбрать(FromDate, ToDate);
			Исключение
				Возврат ОтветJSON(500, "{""ok"":false,""script_version"":""" + JsonEscape(MfSupplierOrdersScriptVersion()) + """,""error"":""select_failed"",""details"":""" + JsonEscape(ОписаниеОшибки()) + """}");
			КонецПопытки;
		КонецЕсли;

		МассивЗаказов = Новый Массив;
		Скан = 0;
		Совп = 0;
		Прерван = Ложь;
		Послед = "";

		Если СписокСсылок <> Неопределено Тогда
			Для Каждого Сс Из СписокСсылок Цикл
				Скан = Скан + 1;
				Если НЕ ЗначениеЗаполнено(Сс) Тогда Продолжить; КонецЕсли;
				Об = Неопределено;
				Попытка Об = Сс.ПолучитьОбъект(); Исключение Об = Неопределено; КонецПопытки;
				Если Об = Неопределено Тогда Продолжить; КонецЕсли;
				Пров = Ложь;
				Попытка Пров = Об.Проведен; Исключение Пров = Ложь; КонецПопытки;
				Если НЕ Пров Тогда Продолжить; КонецЕсли;
				Пом = Ложь;
				Попытка Пом = Об.ПометкаУдаления; Исключение Пом = Ложь; КонецПопытки;
				Если Пом Тогда Продолжить; КонецЕсли;
				Если НЕ SupplierOrderLooksInProgress(Об, КлючСостояния) Тогда Продолжить; КонецЕсли;
				Совп = Совп + 1;
				Если Совп <= Offset Тогда
					Продолжить;
				КонецЕсли;
				Ст = SupplierOrderStateRaw(Об);
				Спр = FindSupplierRef(Об);
				НомерД = "";
				Дт = Неопределено;
				Попытка НомерД = СокрЛП(Об.Номер); Исключение НомерД = ""; КонецПопытки;
				Попытка Дт = Об.Дата; Исключение Дт = ТекущаяДата(); КонецПопытки;
				Од = Новый Структура;
				Од.Вставить("ref", SafeStrForJson(Сс));
				Од.Вставить("uuid", RefUuidString(Сс));
				Од.Вставить("number", НомерД);
				Од.Вставить("date", DateToIso(Дт));
				Од.Вставить("posting", Истина);
				Од.Вставить("state", Новый Структура("name, field, key", Ст.Текст, Ст.Поле, КлючСостояния));
				Од.Вставить("supplier", Спр);
				Од.Вставить("lines", BuildLinesArray(Об));
				МассивЗаказов.Добавить(Од);
				Послед = SafeStrForJson(Сс);
				Если (Limit > 0) И (МассивЗаказов.Количество() >= Limit) Тогда
					Прерван = Истина;
					Прервать;
				КонецЕсли;
			КонецЦикла;
		Иначе
			Пока Выборка.Следующий() Цикл
				Скан = Скан + 1;
				Сс = Неопределено;
				Попытка Сс = Выборка.Ссылка; Исключение Сс = Неопределено; КонецПопытки;
				Если НЕ ЗначениеЗаполнено(Сс) Тогда Продолжить; КонецЕсли;
				Об = Неопределено;
				Попытка Об = Сс.ПолучитьОбъект(); Исключение Об = Неопределено; КонецПопытки;
				Если Об = Неопределено Тогда Продолжить; КонецЕсли;
				Пров = Ложь;
				Попытка Пров = Об.Проведен; Исключение Пров = Ложь; КонецПопытки;
				Если НЕ Пров Тогда Продолжить; КонецЕсли;
				Пом = Ложь;
				Попытка Пом = Об.ПометкаУдаления; Исключение Пом = Ложь; КонецПопытки;
				Если Пом Тогда Продолжить; КонецЕсли;
				Если НЕ SupplierOrderLooksInProgress(Об, КлючСостояния) Тогда Продолжить; КонецЕсли;
				Совп = Совп + 1;
				Если Совп <= Offset Тогда
					Продолжить;
				КонецЕсли;
				Ст = SupplierOrderStateRaw(Об);
				Спр = FindSupplierRef(Об);
				НомерД = "";
				Дт = Неопределено;
				Попытка НомерД = СокрЛП(Об.Номер); Исключение НомерД = ""; КонецПопытки;
				Попытка Дт = Об.Дата; Исключение Дт = ТекущаяДата(); КонецПопытки;
				Од = Новый Структура;
				Од.Вставить("ref", SafeStrForJson(Сс));
				Од.Вставить("uuid", RefUuidString(Сс));
				Од.Вставить("number", НомерД);
				Од.Вставить("date", DateToIso(Дт));
				Од.Вставить("posting", Истина);
				Од.Вставить("state", Новый Структура("name, field, key", Ст.Текст, Ст.Поле, КлючСостояния));
				Од.Вставить("supplier", Спр);
				Од.Вставить("lines", BuildLinesArray(Об));
				МассивЗаказов.Добавить(Од);
				Послед = SafeStrForJson(Сс);
				Если (Limit > 0) И (МассивЗаказов.Количество() >= Limit) Тогда
					Прерван = Истина;
					Прервать;
				КонецЕсли;
			КонецЦикла;
		КонецЕсли;

		Ещё = Ложь;
		Попытка
			Если Прерван Тогда
				Ещё = Истина;
			КонецЕсли;
		Исключение
		КонецПопытки;

		КандН = 0;
		Попытка
			Если СписокСсылок <> Неопределено Тогда
				КандН = СписокСсылок.Количество();
			КонецЕсли;

			КандН = 0;

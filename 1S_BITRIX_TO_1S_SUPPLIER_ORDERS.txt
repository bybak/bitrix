// =============================================================================
// 1C:Enterprise (BSL) — HTTP-обработчик: список проведённых «Заказов поставщикам» в работе
// + ТЧ: номенклатура, количество, price (цена/сумма по строке — см. LinePriceBlock), expected (даты).
//
// quantity: 1) ТЧ (УНФ: КоличествоОсталось, КПоступлению, …) — не «Количество» (это «заказано»);
//   2) Регистр накопления: ЗаказыПоставщикам/ЗаказПоставщикам × измерения ЗаказПоставщику/Заказ/ЗаказПоставщика,
//   варианты с/без кода строки, пока не найдётся К>0; иначе 0 по успешно выполненным запросам. Подмены «заказано» нет.
//   Под настройки: ИмяРесурсаКПоступлению(), KPostupleniyuTableFieldNames(), KPostupleniyuFromRegisterQueryAttempt.
//   Нулевое количество в lines допускается. Даты в JSON — LineExpectedBlock.
//
// Размещение: общий модуль (Сервер, глобальный) — фактическое имя, например MfBitrixUnfSupplierOrdersGet
//             (должно совпадать с вызовом в HTTP-модуле: см. 1S_BITRIX_TO_1S_ORDERS_HANDLER.txt)
// + в публикации HTTP-сервиса: шаблон URL GET → SupplierOrders(Запрос) Экспорт
//   Пример (имя сервиса в метаданных: supplier_order_get, шаблон, напр. list):
//   https://<хост>/unf/hs/supplier_order_get/list
//
// Авторизация: Basic (публикация) + заголовок X-MF-Token (как в 1S_BITRIX_TO_1S_ORDER_CREATE.txt)
// Параметры: X-MF-From / X-MF-To (по умолчанию с 2000-01-01 по «сегодня»);
//   X-MF-Limit: 0 или нет заголовка = все подходящие; >0 = не больше N. X-MF-Offset — пропустить N совпавших.
//
// ВАЖНО: Сверьте в конфигураторе имя документа (часто «ЗаказПоставщику»), ТЧ (Товары/Запасы)
//        и реквизит состояния. Ниже — перебор типовых имён; при расхождении поправьте
//        функции SupplierOrderStateRaw / SupplierOrderLooksInProgress.
// =============================================================================

Функция ParseIsoDateSafe(S) Экспорт
	S = СокрЛП(Строка(S));
	Если ПустаяСтрока(S) Тогда Возврат Неопределено; КонецЕсли;
	П = Найти(S, "+");
	Если П > 0 Тогда S = Лев(S, П - 1); КонецЕсли;
	П = Найти(S, "Z");
	Если П > 0 Тогда S = Лев(S, П - 1); КонецЕсли;
	S = СтрЗаменить(S, "T", " ");
	Если СтрДлина(S) >= 10 И Найти(S, "-") > 0 Тогда
		Попытка
			Год = Число(Лев(S, 4));
			Мес = Число(Сред(S, 6, 2));
			Дн = Число(Сред(S, 9, 2));
			Ч = 0; Мн = 0; Сек = 0;
			Если СтрДлина(S) >= 19 Тогда
				Ч = Число(Сред(S, 12, 2));
				Мн = Число(Сред(S, 15, 2));
				Сек = Число(Сред(S, 18, 2));
			КонецЕсли;
			Возврат Дата(Год, Мес, Дн, Ч, Мн, Сек);
		Исключение
		КонецПопытки;
	КонецЕсли;
	Попытка
		Возврат Дата(S);
	Исключение
		Возврат Неопределено;
	КонецПопытки;

Функция SupplierOrderStateRaw(ДокОбъект) Экспорт

			КонецЕсли;
		Исключение
		КонецПопытки;
	КонецЦикла;
	Возврат Новый Структура("Поле, Текст", "", "");

Функция SupplierOrderLooksInProgress(ДокОбъект, КлючСостояния) Экспорт
	// Ключ: подстрока в нижнем регистре, напр. "в работе" / "вработе" (см. заголовок X-MF-State-Key)
	Ключ = СокрЛП(НРег(Строка(КлючСостояния)));
	Если ПустаяСтрока(Ключ) Тогда Ключ = "в работе"; КонецЕсли;
	Ключ2 = СтрЗаменить(Ключ, " ", "");

	Сыр = SupplierOrderStateRaw(ДокОбъект);
	Т0 = СокрЛП(Сыр.Текст);
	Т = НРег(Т0);
	Тс = СтрЗаменить(Т, " ", "");

	Если Найти(Т, Ключ) > 0 Тогда Возврат Истина; КонецЕсли;
	Если Найти(Тс, Ключ2) > 0 Тогда Возврат Истина; КонецЕсли;
	// Часто в УНФ: "Заказ выполняется" и т.п. — мягкое «работ»
	Если Найти(Т, "работ") > 0 Тогда Возврат Истина; КонецЕсли;
	Возврат Ложь;

// Цена/сумма в строке ТЧ — разные конфигурации: Цена, Сумма, с НДС/без
Функция LinePriceBlock(Стр) Экспорт
	Б = Новый Структура;
	Б.Вставить("unit_price", 0);
	Б.Вставить("line_sum", 0);
	Б.Вставить("sum_with_vat", 0);
	Б.Вставить("vat_rate", "");
	Б.Вставить("unit_price_field", "");
	Б.Вставить("line_sum_field", "");
	// Сначала цена за единицу (приоритет: явная «Цена», затем аналоги)
	КандЦ = СтрРазделить("Цена,ЦенаНоменклатуры,ЦенаБезСкидки,ЦенаСоСкидкой,Себестоимость,Сумма/Количество", ",", Ложь);
	Для Каждого Им0 Из КандЦ Цикл
		Им = СокрЛП(Строка(Им0));
		Если ПустаяСтрока(Им) Тогда Продолжить; КонецЕсли;
		Если Им = "Сумма/Количество" Тогда
			Попытка
				Сум0 = 0; Кол0 = 0;
				Попытка Сум0 = Число(Стр.Сумма); Исключение Сум0 = 0; КонецПопытки;
				Попытка Кол0 = Число(Стр.Количество); Исключение Кол0 = 0; КонецПопытки;
				Если Кол0 = 0 Тогда Попытка Кол0 = Число(Стр.КоличествоУпаковок); Исключение КонецПопытки; КонецЕсли;
				Если Сум0 <> 0 И Кол0 > 0 Тогда
					Б.Вставить("unit_price", Окр(Сум0 / Кол0, 2));
					Б.Вставить("unit_price_field", "Сумма/Количество");
					Прервать;
				КонецЕсли;
			Исключение
			КонецПопытки;
			Продолжить;
		КонецЕсли;
		ЗнД = GetFieldValueSafe(Стр, Им);
		Попытка
			Если ЗнД <> Неопределено Тогда
				Ч0 = 0; Ок = Ложь;
				Если ТипЗнч(ЗнД) = Тип("Число") Тогда
					Ч0 = ЗнД; Ок = Истина;
				Иначе
					Попытка
						Ч0 = Число(ЗнД);
						Ок = Истина;
					Исключение
						Ок = Ложь;
					КонецПопытки;
				КонецЕсли;
				Если Ок Тогда
					Б.Вставить("unit_price", Окр(Ч0, 4));
					Б.Вставить("unit_price_field", Им);
					Прервать;
				КонецЕсли;
			КонецЕсли;
		Исключение
		КонецПопытки;
	КонецЦикла;
	// Сумма по строке (не путать с суммой НДС)

Функция SupplierOrders(Запрос) Экспорт
	// Сигнатура для HTTP: Функция(HTTPСервисЗапрос) -> HTTPСервисОтвет

		// --- token ---
		Ож = СокрЛП(MfSupplierOrdersExpectedToken());

			Ток = СокрЛП(HeaderValue(Запрос, "X-MF-Token"));
			Если Ток <> Ож Тогда
				Возврат ОтветJSON(401, "{""ok"":false,""error"":""unauthorized""}");
			КонецЕсли;
		КонецЕсли;

		// --- debug version ---
		Дб = СокрЛП(ВРег(HeaderValue(Запрос, "X-MF-Debug")));
		Если (Дб = "VER") Или (Дб = "VERSION") Тогда
			С0 = Новый Структура("ok, script_version", Истина, MfSupplierOrdersScriptVersion());
			Возврат ОтветJSON(200, ЗаписатьСтруктуруВJSON(С0));
		КонецЕсли;

		// --- период: по умолчанию вся история; X-MF-From / X-MF-To сузить ---
		FromDef = Дата(2000, 1, 1);
		ToDef = КонецДня(ТекущаяДата());
		FromDate = ParseDateHeaderSafe(Запрос, "X-MF-From", FromDef);
		ToDate = ParseDateHeaderSafe(Запрос, "X-MF-To", ToDef);
		Попытка ToDate = КонецДня(ToDate); Исключение КонецПопытки;
		Попытка FromDate = НачалоДня(FromDate); Исключение КонецПопытки;

		// --- X-MF-Offset: пропуск N подходящих; X-MF-Limit: 0 (по умолчанию) = без лимита, >0 = не больше N ---
		Offset = ParseIntSafe(HeaderValue(Запрос, "X-MF-Offset"), 0);
		Limit = ParseIntSafe(HeaderValue(Запрос, "X-MF-Limit"), 0);
		Если Limit < 0 Тогда Limit = 0; КонецЕсли;
		Если Offset < 0 Тогда Offset = 0; КонецЕсли;

		КлючСостояния = СокрЛП(HeaderValue(Запрос, "X-MF-State-Key"));
		Если ПустаяСтрока(КлючСостояния) Тогда КлючСостояния = "в работе"; КонецЕсли;

		// --- менеджер (для fallback) + запрос «новейшие по дате» (основной путь) ---
		Мен = TryGetDocManagerByName("ЗаказПоставщику");

		Если Мен = Неопределено Тогда
			Возврат ОтветJSON(500, "{""ok"":false,""script_version"":""" + JsonEscape(MfSupplierOrdersScriptVersion()) + """,""error"":""doc_manager_not_found""}");
		КонецЕсли;

		// 0 = в запросе без «ПЕРВЫХ» (все проведённые в период), дальше отсекаем «не в работе» в цикле
		СписокСсылок = НовейшиеСсылкиПроведённыхПоПериоду(FromDate, ToDate, 0);
		РежимВыборки = "query_newest";
		Выборка = Неопределено;
		Если СписокСсылок = Неопределено Тогда
			РежимВыборки = "manager_select";
			Попытка
				Выборка = Мен.Выбрать(FromDate, ToDate);
			Исключение
				Возврат ОтветJSON(500, "{""ok"":false,""script_version"":""" + JsonEscape(MfSupplierOrdersScriptVersion()) + """,""error"":""select_failed"",""details"":""" + JsonEscape(ОписаниеОшибки()) + """}");

		Совп = 0;
		Прерван = Ложь;
		Послед = "";

			Для Каждого Сс Из СписокСсылок Цикл
				Скан = Скан + 1;
				Если НЕ ЗначениеЗаполнено(Сс) Тогда Продолжить; КонецЕсли;
				Об = Неопределено;
				Попытка Об = Сс.ПолучитьОбъект(); Исключение Об = Неопределено; КонецПопытки;
				Если Об = Неопределено Тогда Продолжить; КонецЕсли;

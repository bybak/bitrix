// 1C:Enterprise (BSL) — пример отдельного HTTP-обработчика для обновления заказа из Bitrix
// Назначение:
// - принимать POST JSON от Bitrix при событии оплаты (meta.event = "payment_paid")
// - логировать входящее тело и результат
// - находить существующий "Заказ покупателя" по order.external_id в комментарии (формат "{USER_ID}-{ORDER_ID}", напр. "1-115")
// - обновлять реквизиты, необходимые для появления вкладки "Оплата"
// - (опционально) создавать документ оплаты и проводить его, чтобы в списке заказов загорелась "зелёная монетка"
//
// ВАЖНО:
// 1) Этот код нужно разместить в модуле HTTP-сервиса (Общий модуль / модуль HTTP-сервиса, зависит от вашей конфигурации).
// 2) Имена метаданных документов/табличных частей могут отличаться в вашей УНФ. Ниже используются типовые:
//    - Документы.ЗаказПокупателя
//    - Документы.ПоступлениеНаРасчетныйСчет
// 3) Идемпотентность оплаты сделана по payment_event.ps_invoice_id (UUID PayKeeper). Если у вас другое поле — замените.
//
// Рекомендуемый URL для Bitrix: /unf/hs/orders/update  (пример)
// В Bitrix достаточно задать MF_UNF_PAID_ENDPOINT на этот URL.

Процедура EnsureOrderSumsNonZero(ЗаказОбъект, PreferredSum) Экспорт
	Если ЗаказОбъект = Неопределено Тогда Возврат; КонецЕсли;
	Если PreferredSum = Неопределено Тогда PreferredSum = 0; КонецЕсли;

	// Важно: в разных доработках "итого" может храниться в разных реквизитах.
	// Если заполнен только один (например СуммаДокумента), а другой (Сумма/Итого) = 0,
	// UI может показывать 0. Поэтому выходим только если оба уже > 0.
	СумДок0 = 0;
	Сум0 = 0;
	Попытка СумДок0 = ЗаказОбъект.СуммаДокумента; Исключение СумДок0 = 0; КонецПопытки;
	Попытка Сум0 = ЗаказОбъект.Сумма; Исключение Сум0 = 0; КонецПопытки;
	Если (СумДок0 > 0) И (Сум0 > 0) Тогда Возврат; КонецЕсли;

	СтавкаБезНДС = VatRateBezNdsRef();
	СуммаПоСтрокам = 0;

	// В разных доработках УНФ табличная часть товаров может называться по-разному.
	КандидатыТЧ = Новый Массив;
	КандидатыТЧ.Добавить("Запасы");
	КандидатыТЧ.Добавить("Товары");
	КандидатыТЧ.Добавить("ТоварыУслуги");
	КандидатыТЧ.Добавить("ТоварыИУслуги");
	КандидатыТЧ.Добавить("ТоварыРаботыУслуги");

	Для Каждого ИмяТЧ Из КандидатыТЧ Цикл
		ТЧ = TryGetTablePart(ЗаказОбъект, ИмяТЧ);
		Если ТЧ = Неопределено Тогда Продолжить; КонецЕсли;

		Перв = Истина;
		Для Каждого Стр Из ТЧ Цикл
			Кол = 0; Цена = 0; Сум = 0;
			Попытка Кол = Стр.Количество; Исключение Кол = 0; КонецПопытки;
			Если Кол = 0 Тогда Попытка Кол = Стр.Количествo; Исключение Кол = 0; КонецПопытки; КонецЕсли; // иногда встречается с опечаткой в доработках
			Если Кол = 0 Тогда Попытка Кол = Стр.КоличествоУпаковок; Исключение Кол = 0; КонецПопытки; КонецЕсли;
			Попытка Цена = Стр.Цена; Исключение Цена = 0; КонецПопытки;
			Если Цена = 0 Тогда Попытка Цена = Стр.ЦенаПродажи; Исключение Цена = 0; КонецПопытки; КонецЕсли;
			Если Цена = 0 Тогда Попытка Цена = Стр.ЦенаЕдиницы; Исключение Цена = 0; КонецПопытки; КонецЕсли;

			// сумма строки может храниться в разных полях
			Попытка Сум = Стр.Сумма; Исключение Сум = 0; КонецПопытки;
			Если Сум = 0 Тогда
				Попытка Сум = Стр.СуммаСтроки; Исключение Сум = 0; КонецПопытки;
			КонецЕсли;
			Если Сум = 0 Тогда
				Сум = Кол * Цена;
				Если Сум <> 0 Тогда
					Попытка Стр.Сумма = Сум; Исключение КонецПопытки;
					Попытка Стр.СуммаСтроки = Сум; Исключение КонецПопытки;
				КонецЕсли;
			КонецЕсли;

			// Если вообще нули (часто при некорректном маппинге цены) — дожмем первую строку из PreferredSum.
			Если (Сум = 0) И (PreferredSum > 0) И Перв Тогда
				Кол = 1; Цена = PreferredSum; Сум = PreferredSum;
				Попытка Стр.Количество = 1; Исключение КонецПопытки;
				Попытка Стр.КоличествоУпаковок = 1; Исключение КонецПопытки;
				Попытка Стр.Цена = PreferredSum; Исключение КонецПопытки;
				Попытка Стр.ЦенаПродажи = PreferredSum; Исключение КонецПопытки;
				Попытка Стр.Сумма = PreferredSum; Исключение КонецПопытки;
				Попытка Стр.СуммаСтроки = PreferredSum; Исключение КонецПопытки;
			КонецЕсли;

			Если СтавкаБезНДС <> Неопределено Тогда
				Попытка Стр.СтавкаНДС = СтавкаБезНДС; Исключение КонецПопытки;
			КонецЕсли;
			Попытка Стр.СуммаНДС = 0; Исключение КонецПопытки;
			СуммаПоСтрокам = СуммаПоСтрокам + Сум;
			Перв = Ложь;
		КонецЦикла;
	КонецЦикла;

	Если (СуммаПоСтрокам = 0) И (PreferredSum > 0) Тогда
		СуммаПоСтрокам = PreferredSum;
	КонецЕсли;
	Если СуммаПоСтрокам > 0 Тогда
		Попытка ЗаказОбъект.СуммаДокумента = СуммаПоСтрокам; Исключение КонецПопытки;
		Попытка ЗаказОбъект.Сумма = СуммаПоСтрокам; Исключение КонецПопытки;
		Попытка ЗаказОбъект.Итого = СуммаПоСтрокам; Исключение КонецПопытки;
		Попытка ЗаказОбъект.СуммаВсего = СуммаПоСтрокам; Исключение КонецПопытки;
		Попытка ЗаказОбъект.СуммаКОплате = СуммаПоСтрокам; Исключение КонецПопытки;
		// На некоторых базах итоговые реквизиты пересчитываются только явным вызовом.
		Попытка ЗаказОбъект.ПересчитатьИтоги(); Исключение КонецПопытки;
		Попытка ЗаказОбъект.РассчитатьИтоги(); Исключение КонецПопытки;
	КонецЕсли;
КонецПроцедуры

			"available,description,module,line,source,stack,raw",

		Если Р.module = "" Тогда Попытка Если Инф.Свойство("Модуль", З) Тогда Р.module = SafeStrForJson(З); КонецЕсли; Исключение КонецПопытки; КонецЕсли;
		Если Р.line = "" Тогда Попытка Если Инф.Свойство("Строка", З) Тогда Р.line = SafeStrForJson(З); КонецЕсли; Исключение КонецПопытки; КонецЕсли;
		Если Р.source = "" Тогда Попытка Если Инф.Свойство("ИсходныйТекст", З) Тогда Р.source = SafeStrForJson(З); КонецЕсли; Исключение КонецПопытки; КонецЕсли;
		Если Р.stack = "" Тогда Попытка Если Инф.Свойство("Stack", З) Тогда Р.stack = SafeStrForJson(З); КонецЕсли; Исключение КонецПопытки; КонецЕсли;

		Возврат Р;
	Исключение
		Возврат Новый Структура("available,error", Ложь, SafeStrForJson(ОписаниеОшибки()));
	КонецПопытки;

Процедура EnsureOrderPrepaymentOffsetManual(ЗаказОбъект) Экспорт
	Если ЗаказОбъект = Неопределено Тогда Возврат; КонецЕсли;
	// В режиме "Автоматически" УНФ может очищать табличную часть "Предоплата" при записи/проведении,
	// из-за чего вкладка "Оплата" в заказе показывает 0% даже при разнесенном документе оплаты.
	// Делаем как у менеджеров: "Вручную".
	Попытка
		Попытка ЗаказОбъект.УчестьПредоплату = Истина; Исключение КонецПопытки;
		Попытка ЗаказОбъект.УчитыватьПредоплату = Истина; Исключение КонецПопытки;

		Попытка
			ЗаказОбъект.СпособЗачетаПредоплаты = Перечисления.СпособыЗачетаИРаспределенияПлатежей.Вручную;
		Исключение
			Попытка ЗаказОбъект.СпособЗачетаПредоплаты = Перечисления.СпособыЗачетаИРаспределенияПлатежей.Ручной; Исключение КонецПопытки;
			Попытка ЗаказОбъект.СпособЗачетаАванса = Перечисления.СпособыЗачетаИРаспределенияПлатежей.Вручную; Исключение КонецПопытки;
		КонецПопытки;
	Исключение
	КонецПопытки;

&НаСервере
Функция UpdateOrder(Запрос) Экспорт
	// Для HTTP-сервиса 1С корректная сигнатура обработчика — Функция(Запрос) -> HTTPСервисОтвет.

	Попытка
		// DEBUG: allow checking deployment/version without JSON body
		DebugMode = DebugModeFromRequest(Запрос);
		Если DebugMode = "VER" Тогда
			Возврат ОтветJSON(200, "{""ok"":true,""script_version"":""" + MfUpdateScriptVersion() + """}");
		ИначеЕсли DebugMode = "PAYMETA" Тогда
			Возврат DebugDescribePaymentDoc();
		ИначеЕсли DebugMode = "REPOSTPAY" Тогда
			Возврат DebugRepostPaymentsFromRequest(Запрос);
		ИначеЕсли DebugMode = "FIXPAYSTATUS" Тогда
			Возврат DebugFixPayStatusFromRequest(Запрос);
		КонецЕсли;

		Тело = ПолучитьТелоКакСтроку(Запрос);
		Лог("IN", Тело);

		Вход = ПрочитатьJSONВСтруктуру(Тело);
		Если Вход = Неопределено Тогда
			Возврат ОтветJSON(400, "{""ok"":false,""error"":""bad_json""}");
		КонецЕсли;

		// Требуем order.external_id
		Если НЕ Вход.Свойство("order") Или НЕ Вход.order.Свойство("external_id") Тогда
			Возврат ОтветJSON(400, "{""ok"":false,""error"":""no_external_id""}");
		КонецЕсли;

		ВнешнийИД = "" + Вход.order.external_id;
		Событие = "";
		Если Вход.Свойство("meta") И Вход.meta.Свойство("event") Тогда
			Событие = "" + Вход.meta.event;
		КонецЕсли;

		// 1) Найти заказ (по комментарию "external_id: {USER_ID}-{ORDER_ID}")
		ЗаказСсылка = НайтиЗаказПоВнешнемуИД(ВнешнийИД);
		Если ЗаказСсылка = Неопределено Тогда
			Возврат ОтветJSON(404, "{""ok"":false,""script_version"":""" + MfUpdateScriptVersion() + """,""error"":""order_not_found""}");
		КонецЕсли;

		ЗаказОбъект = ЗаказСсылка.ПолучитьОбъект();

		// 2) Подготовить реквизиты для вкладки "Оплата"
		ОбеспечитьРеквизитыОплаты(ЗаказОбъект, Вход);

		// Диагностика: что именно получилось заполнить для вкладки "Оплата"
		Diag = Новый Структура;
		Попытка
			// реквизиты
			Планировать = Неопределено;
			Попытка Планировать = ЗаказОбъект.ЗапланироватьОплату; Исключение Планировать = Неопределено; КонецПопытки;
			Diag.Вставить("ЗапланироватьОплату", Планировать);

			ИдПлат = "";
			Попытка ИдПлат = "" + ЗаказОбъект.ИдентификаторПлатежа; Исключение ИдПлат = ""; КонецПопытки;
			Diag.Вставить("ИдентификаторПлатежа", ИдПлат);

			// табчасти
			КолКал = 0;
			Попытка КолКал = ЗаказОбъект.ПлатежныйКалендарь.Количество(); Исключение КолКал = 0; КонецПопытки;
			Diag.Вставить("ПлатежныйКалендарь_count", КолКал);

			КолПред = 0;
			Попытка КолПред = ЗаказОбъект.Предоплата.Количество(); Исключение КолПред = 0; КонецПопытки;
			Diag.Вставить("Предоплата_count", КолПред);
		Исключение
		КонецПопытки;

		// 3) Если это событие оплаты — создать документ оплаты и провести
		Если Событие = "payment_paid" Тогда
			// Для PayKeeper-оплаты ставим ручной зачет, как у менеджеров (иначе УНФ может чистить "Предоплата")
			EnsureOrderPrepaymentOffsetManual(ЗаказОбъект);
			// Если заказ с нулевой суммой — оплата выглядит как переплата ("мы должны"). Чиним только если суммы реально 0.
			СуммаОпл0 = 0;
			Попытка СуммаОпл0 = Вход.payment_event.sum; Исключение СуммаОпл0 = 0; КонецПопытки;
			Если СуммаОпл0 = 0 Тогда Попытка СуммаОпл0 = Вход.order.price; Исключение СуммаОпл0 = 0; КонецПопытки; КонецЕсли;
			Попытка EnsureOrderSumsNonZero(ЗаказОбъект, СуммаОпл0); Исключение КонецПопытки;
			РезОплата = ПровестиОплатуПоЗаказу(ЗаказОбъект, Вход);
			// ВАЖНО: "закрашенный кружок" в списке заказов УНФ считается по РС ФактОплатыЗаказов/ГрафикОплатыЗаказов.
			// Документ оплаты и табчасть "Предоплата" в заказе могут быть заполнены, но индикатор в списке не обновиться,
			// пока не будет корректных записей в этих регистрах. Заполняем их ТОЛЬКО для текущего заказа.
			Попытка

Функция DebugRepostPaymentsFromRequest(Запрос) Экспорт
	// Пакетное перепроведение документов для восстановления статусов оплат.
	// Управляется заголовками:
	// - X-MF-Doc: ORDER|INVOICE|CASH|BANK
	// - X-MF-From: 2000-01-01
	// - X-MF-To: 2100-01-01
	// - X-MF-Offset: 0
	// - X-MF-Limit: 200 (max 500)
	// - X-MF-Mode: POST|WRITE

	FromDef = Дата(2000, 1, 1);
	ToDef = КонецДня(ТекущаяДата());
	FromDate = ParseDateHeaderSafe(Запрос, "X-MF-From", FromDef);
	ToDate = ParseDateHeaderSafe(Запрос, "X-MF-To", ToDef);
	// Чтобы "2026-03-12" включал весь день, а не 00:00:00
	Попытка ToDate = КонецДня(ToDate); Исключение КонецПопытки;
	Попытка FromDate = НачалоДня(FromDate); Исключение КонецПопытки;
	Offset = ParseIntSafe(HeaderValue(Запрос, "X-MF-Offset"), 0);
	Limit = ParseIntSafe(HeaderValue(Запрос, "X-MF-Limit"), 200);
	Mode = СокрЛП(HeaderValue(Запрос, "X-MF-Mode"));
	DocType = ВРег(СокрЛП(HeaderValue(Запрос, "X-MF-Doc")));
	Если ПустаяСтрока(DocType) Тогда DocType = "CASH"; КонецЕсли;

	Мен = Неопределено;
	Имя = "";
	Если DocType = "ORDER" Тогда
		Имя = "ЗаказПокупателя";
	ИначеЕсли DocType = "INVOICE" Тогда
		Имя = "СчетНаОплату";
	ИначеЕсли DocType = "CASH" Тогда
		Имя = "ПоступлениеВКассу";
	ИначеЕсли DocType = "BANK" Тогда
		// пробуем распространенные имена
		Мен = TryGetDocManagerByName("ПоступлениеНаРасчетныйСчет");
		Если Мен = Неопределено Тогда Мен = TryGetDocManagerByName("ПоступлениеНаСчет"); КонецЕсли;
		Если Мен = Неопределено Тогда Мен = TryGetDocManagerByName("ПоступлениеНаРасчетный"); КонецЕсли;
	Иначе
		Возврат ОтветJSON(400, "{""ok"":false,""script_version"":""" + MfUpdateScriptVersion() + """,""error"":""bad_doc_type"",""hint"":""use X-MF-Doc: ORDER|INVOICE|CASH|BANK""}");
	КонецЕсли;

	Если Мен = Неопределено И НЕ ПустаяСтрока(Имя) Тогда
		Мен = TryGetDocManagerByName(Имя);
	КонецЕсли;
	Если Мен = Неопределено Тогда
		Возврат ОтветJSON(404, "{""ok"":false,""script_version"":""" + MfUpdateScriptVersion() + """,""error"":""doc_manager_not_found"",""doc"":""" + DocType + """}");
	КонецЕсли;

	Р = RepostDocumentsByManager(Мен, FromDate, ToDate, Offset, Limit, Mode);

	Out = Новый Структура(
		"ok,script_version,debug,doc,from,to,offset,limit,mode,result",
		Истина, MfUpdateScriptVersion(), "repost_payments", DocType, Строка(FromDate), Строка(ToDate), Offset, Limit, Mode, Р
	);
	Возврат ОтветJSON(200, ЗаписатьСтруктуруВJSON(Out));

Процедура ОбеспечитьРеквизитыОплаты(ЗаказОбъект, Вход) Экспорт
	// Идея: вкладка "Оплата" в УНФ часто зависит от того, заполнены ли:
	// - Организация
	// - Контрагент / Договор
	// - Условия оплаты / План оплат (табличная часть)
	//
	// Если у вас эти поля уже заполняются в create-обработчике — эта процедура ничего не испортит (идемпотентна).

	// Организация (пример: берём как уже заданную или ставим "первую попавшуюся")
	Если ЗаказОбъект.Организация = Неопределено Тогда
		// Замените на вашу конкретную организацию/логику
		Попытка
			Выборка = Справочники.Организации.Выбрать();
			Если Выборка.Следующий() Тогда
				ЗаказОбъект.Организация = Выборка.Ссылка;
			КонецЕсли;
		Исключение
		КонецПопытки;
	КонецЕсли;

	// Контрагент/Договор (зависит от конфигурации)
	// Если у вас в ЗаказПокупателя нет этих реквизитов — просто удалите блок.
	Попытка
		// Контрагент
		ЗнКонтр = Неопределено;
		Попытка ЗнКонтр = ЗаказОбъект.Контрагент; Исключение ЗнКонтр = Неопределено; КонецПопытки;
		Если ЗнКонтр = Неопределено Тогда
			// Пытаемся получить email/телефон/ФИО и найти/создать контрагента.
			Имя = "";
			Если Вход.order.Свойство("customer") И Вход.order.customer.Свойство("name") Тогда
				Имя = "" + Вход.order.customer.name;
			КонецЕсли;
			НовыйКонтр = НайтиИлиСоздатьКонтрагента(Имя, Вход);
			Если НовыйКонтр <> Неопределено Тогда
				Попытка ЗаказОбъект.Контрагент = НовыйКонтр; Исключение КонецПопытки;
				Попытка ЗаказОбъект.Покупатель = НовыйКонтр; Исключение КонецПопытки;
			КонецЕсли;
		КонецЕсли;

		// Договор: по вашей META есть реквизит "Договор" (а "ДоговорКонтрагента" может отсутствовать).
		ЗнДог = Неопределено;
		Попытка ЗнДог = ЗаказОбъект.Договор; Исключение ЗнДог = Неопределено; КонецПопытки;
		Если ЗнДог = Неопределено Тогда
			ЗнКонтр2 = Неопределено;
			Попытка ЗнКонтр2 = ЗаказОбъект.Контрагент; Исключение ЗнКонтр2 = Неопределено; КонецПопытки;
			ЗнОрг2 = Неопределено;
			Попытка ЗнОрг2 = ЗаказОбъект.Организация; Исключение ЗнОрг2 = Неопределено; КонецПопытки;
			Если ЗнКонтр2 <> Неопределено Тогда
				НД = НайтиИлиСоздатьДоговор(ЗнКонтр2, ЗнОрг2);
				Если НД <> Неопределено Тогда
					Попытка ЗаказОбъект.Договор = НД; Исключение КонецПопытки;
					// на всякий случай, если реквизит всё же есть
					Попытка ЗаказОбъект.ДоговорКонтрагента = НД; Исключение КонецПопытки;
				КонецЕсли;
			КонецЕсли;
		КонецЕсли;
	Исключение
	КонецПопытки;

	// Явный флаг из вашей меты: "Запланировать оплату"
	Попытка

Процедура EnsurePaymentDocBankAccounts(ДокОплаты, ЗаказОбъект) Экспорт
	Если ДокОплаты = Неопределено Тогда Возврат; КонецЕсли;

	// БанковскийСчет (организация)
	Попытка
		Тек = Неопределено;
		Попытка Тек = ДокОплаты.БанковскийСчет; Исключение Тек = Неопределено; КонецПопытки;
		// Важно: если счет уже заполнен, но у него пустой СчетУчета, проведение падает "Не заполнен счет дебета".
		НужноПодобрать = (НЕ ЗначениеЗаполнено(Тек));
		Если (НЕ НужноПодобрать) Тогда
			Попытка
				НужноПодобрать = НЕ BankAccountHasAccounting(Тек);
			Исключение
				НужноПодобрать = Ложь;
			КонецПопытки;
		КонецЕсли;

		Если НужноПодобрать Тогда
			СчетИзЗаказа = Неопределено;
			Если ЗаказОбъект <> Неопределено Тогда
				Попытка СчетИзЗаказа = ЗаказОбъект.БанковскийСчет; Исключение СчетИзЗаказа = Неопределено; КонецПопытки;
			КонецЕсли;
			Если ЗначениеЗаполнено(СчетИзЗаказа) И BankAccountHasAccounting(СчетИзЗаказа) Тогда
				Попытка ДокОплаты.БанковскийСчет = СчетИзЗаказа; Исключение КонецПопытки;
			Иначе
				Орг = Неопределено;
				Попытка Орг = ДокОплаты.Организация; Исключение Орг = Неопределено; КонецПопытки;
				Если НЕ ЗначениеЗаполнено(Орг) И (ЗаказОбъект <> Неопределено) Тогда
					Попытка Орг = ЗаказОбъект.Организация; Исключение Орг = Неопределено; КонецПопытки;
				КонецЕсли;
				СчетОрг = BankAccountByOwnerWithAccounting(Орг);
				Если НЕ ЗначениеЗаполнено(СчетОрг) Тогда
					СчетОрг = FirstBankAccountByOwner(Орг);
				КонецЕсли;
				Если НЕ ЗначениеЗаполнено(СчетОрг) Тогда
					// Если у организации нет банковских счетов с учетом, попробуем любой счет с заполненным СчетУчета.
					СчетОрг = AnyBankAccountWithAccounting();
				КонецЕсли;
				Если ЗначениеЗаполнено(СчетОрг) Тогда
					Попытка ДокОплаты.БанковскийСчет = СчетОрг; Исключение КонецПопытки;
				КонецЕсли;
			КонецЕсли;
		КонецЕсли;
	Исключение
	КонецПопытки;

	// СчетКонтрагента (плательщик)
	Попытка
		Тек2 = Неопределено;
		Попытка Тек2 = ДокОплаты.СчетКонтрагента; Исключение Тек2 = Неопределено; КонецПопытки;
		Если НЕ ЗначениеЗаполнено(Тек2) Тогда
			Контр = Неопределено;
			Попытка Контр = ДокОплаты.Контрагент; Исключение Контр = Неопределено; КонецПопытки;
			Если НЕ ЗначениеЗаполнено(Контр) И (ЗаказОбъект <> Неопределено) Тогда
				Попытка Контр = ЗаказОбъект.Контрагент; Исключение Контр = Неопределено; КонецПопытки;
			КонецЕсли;
			СчетКонтр = BankAccountByOwnerWithAccounting(Контр);
			Если НЕ ЗначениеЗаполнено(СчетКонтр) Тогда
				СчетКонтр = FirstBankAccountByOwner(Контр);
			КонецЕсли;
			Если ЗначениеЗаполнено(СчетКонтр) Тогда
				Попытка ДокОплаты.СчетКонтрагента = СчетКонтр; Исключение КонецПопытки;
			Иначе
				// крайний fallback: если контрагентский счет отсутствует, но нужен для проведения,
				// попробуем подставить счет организации, чтобы не блокировать проведение.
				СчетОрг2 = Неопределено;
				Попытка СчетОрг2 = ДокОплаты.БанковскийСчет; Исключение СчетОрг2 = Неопределено; КонецПопытки;
				Если ЗначениеЗаполнено(СчетОрг2) Тогда
					Попытка ДокОплаты.СчетКонтрагента = СчетОрг2; Исключение КонецПопытки;
				Иначе
					// ещё один fallback: любой счет с учетом
					СчетЛюб = AnyBankAccountWithAccounting();
					Если ЗначениеЗаполнено(СчетЛюб) Тогда
						Попытка ДокОплаты.СчетКонтрагента = СчетЛюб; Исключение КонецПопытки;
					КонецЕсли;
				КонецЕсли;
			КонецЕсли;
		КонецЕсли;
	Исключение
	КонецПопытки;

		// Оставляем одну строку, чтобы сумма документа = сумме разнесенных платежей.

	Если ДокОплаты = Неопределено ИЛИ ЗаказСсылка = Неопределено Тогда Возврат Рез; КонецЕсли;

	МД = Неопределено;
	Попытка МД = ДокОплаты.Метаданные(); Исключение МД = Неопределено; КонецПопытки;
	Если МД = Неопределено Тогда Возврат Рез; КонецЕсли;

	// 1) Пробуем шапку документа
	Попытка
		Для Каждого Р Из МД.Реквизиты Цикл
			Типы = Неопределено;
			Попытка Типы = Р.Тип.Типы(); Исключение Типы = Неопределено; КонецПопытки;
			Если Типы = Неопределено Тогда Продолжить; КонецЕсли;
			Для Каждого Т Из Типы Цикл
				Если Найти(Строка(Т), "ДокументСсылка.ЗаказПокупателя") > 0 Тогда
					Попытка
						Выполнить("ДокОплаты." + Р.Имя + " = ЗаказСсылка;");

						Рез.where = "header";
						Рез.field = Р.Имя;
						Возврат Рез;
					Исключение
					КонецПопытки;
				КонецЕсли;
			КонецЦикла;
		КонецЦикла;
	Исключение
	КонецПопытки;

	// Важно: не создаем строки в табчастях, чтобы не плодить "Разнесение платежа".
	// Связь с заказом обеспечивается в EnsurePaymentBreakdownRow (он очищает табчасть и создаёт одну корректную строку).
	Возврат Рез;

Функция ParseIsoDateSafe(S) Экспорт
	// Принимаем строки вида "2026-03-13T00:04:43+03:00" / "...Z" и возвращаем Дата или Неопределено
	S = СокрЛП(Строка(S));
	Если ПустаяСтрока(S) Тогда Возврат Неопределено; КонецЕсли;

	// Отрезаем таймзону
	П = Найти(S, "+");
	Если П > 0 Тогда S = Лев(S, П - 1); КонецЕсли;
	П = Найти(S, "Z");
	Если П > 0 Тогда S = Лев(S, П - 1); КонецЕсли;

	S = СтрЗаменить(S, "T", " ");

	// ISO "YYYY-MM-DD ..." может не парситься функцией Дата() в русской локали — разберём вручную.
	Если СтрДлина(S) >= 10 И Найти(S, "-") > 0 Тогда
		Попытка
			Год = Число(Лев(S, 4));
			Мес = Число(Сред(S, 6, 2));
			Дн = Число(Сред(S, 9, 2));
			Ч = 0; Мн = 0; Сек = 0;
			Если СтрДлина(S) >= 19 Тогда
				Ч = Число(Сред(S, 12, 2));
				Мн = Число(Сред(S, 15, 2));
				Сек = Число(Сред(S, 18, 2));
			КонецЕсли;
			Возврат Дата(Год, Мес, Дн, Ч, Мн, Сек);
		Исключение
		КонецПопытки;
	КонецЕсли;

	Попытка
		Возврат Дата(S);
	Исключение
		Возврат Неопределено;
	КонецПопытки;

Процедура EnsurePaymentIndicatorRegistersPaid(ЗаказСсылка, СуммаДокумента, СуммаОплаты, ДатаПлатежа, Dbg = Неопределено) Экспорт
	// Заполняем РС ФактОплатыЗаказов / ГрафикОплатыЗаказов для корректного расчета "Всего оплачено" в заказе.
	Если НЕ ЗначениеЗаполнено(ЗаказСсылка) Тогда Возврат; КонецЕсли;
	Если СуммаДокумента = Неопределено Тогда СуммаДокумента = 0; КонецЕсли;
	Если СуммаОплаты = Неопределено Тогда СуммаОплаты = 0; КонецЕсли;
	Если ДатаПлатежа = Неопределено Тогда ДатаПлатежа = ТекущаяДата(); КонецЕсли;
	Если СуммаДокумента = 0 Тогда СуммаДокумента = СуммаОплаты; КонецЕсли;

	Если Dbg <> Неопределено Тогда
		Попытка Dbg.Вставить("order_ref", Строка(ЗаказСсылка)); Исключение КонецПопытки;
		Попытка Dbg.Вставить("sum_order", СуммаДокумента); Исключение КонецПопытки;
		Попытка Dbg.Вставить("sum_paid", СуммаОплаты); Исключение КонецПопытки;
		Попытка Dbg.Вставить("paid_at", Строка(ДатаПлатежа)); Исключение КонецПопытки;
	КонецЕсли;

	// В разных доработках измерение регистра может называться по-разному.
	Кандидаты = Новый Массив;
	Кандидаты.Добавить("СчетНаОплату");
	Кандидаты.Добавить("Заказ");
	Кандидаты.Добавить("ЗаказПокупателя");
	Кандидаты.Добавить("Документ");
	Кандидаты.Добавить("ДокументОснование");
	DimFact = RegPickMeasurementNameByCandidates("ФактОплатыЗаказов", Кандидаты);
	DimPlan = RegPickMeasurementNameByCandidates("ГрафикОплатыЗаказов", Кандидаты);
	Если Dbg <> Неопределено Тогда
		Попытка Dbg.Вставить("dim_fact", DimFact); Исключение КонецПопытки;
		Попытка Dbg.Вставить("dim_plan", DimPlan); Исключение КонецПопытки;
	КонецЕсли;

	// 1) Факт оплаты заказов
	Попытка
		НФ = РегистрыСведений.ФактОплатыЗаказов.СоздатьНаборЗаписей();
		Если НЕ ПустаяСтрока(DimFact) Тогда
			TrySetFilterByName(НФ, DimFact, ЗаказСсылка);
		Иначе
			// без фильтра писать нельзя
			Если Dbg <> Неопределено Тогда Попытка Dbg.Вставить("fact_write_error", "no_dimension"); Исключение КонецПопытки; КонецЕсли;

&НаСервере
Функция EnsureInvoiceForOrder(ЗаказСсылка, PreferredSum) Экспорт
	// Вариант "как оплата заказа": создаем/находим СчетНаОплату по основанию (Заказ),
	// чтобы платеж закрывал долг по счету (и не было "мы должны/переплата" из-за аванса).
	// При этом в оплате остаётся фискальная метка "предоплата" (ПризнакСпособаРасчета=ПредоплатаПолная).
	Р = Новый Структура("ok,posted,ref,error", Ложь, Ложь, Неопределено, "");
	Если НЕ ЗначениеЗаполнено(ЗаказСсылка) Тогда Р.error = "no_order_ref"; Возврат Р; КонецЕсли;
	Если PreferredSum = Неопределено Тогда PreferredSum = 0; КонецЕсли;

	// 1) Попробуем найти существующий счет
	Счет = Неопределено;
	Попытка
		Счет = УчетРасчетовСКонтрагентами.ПолучитьСчетНаОплатуПоОснованию(ЗаказСсылка, Истина);
	Исключение
		Счет = Неопределено;
	КонецПопытки;
	Если ЗначениеЗаполнено(Счет) Тогда
		// Игнорируем удалённые/нулевые счета и по возможности проводим/чинем существующие
		СчетОбъект = Неопределено;
		Попытка СчетОбъект = Счет.ПолучитьОбъект(); Исключение СчетОбъект = Неопределено; КонецПопытки;
		Если СчетОбъект <> Неопределено Тогда
			ПомУдал = Ложь;
			Попытка ПомУдал = СчетОбъект.ПометкаУдаления; Исключение ПомУдал = Ложь; КонецПопытки;
			Если НЕ ПомУдал Тогда
				НужноПерепровести = Ложь;

				// 1) Сумма счета: если 0 — проставляем PreferredSum (и по необходимости правим строки)
				СуммаСчета = 0;
				Попытка СуммаСчета = СчетОбъект.СуммаДокумента; Исключение СуммаСчета = 0; КонецПопытки;
				Если (СуммаСчета = 0) И (PreferredSum > 0) Тогда
					Попытка СчетОбъект.СуммаДокумента = PreferredSum; Исключение КонецПопытки;
					НужноПерепровести = Истина;
				КонецЕсли;

				// 2) Ставка НДС в строках счета обязательна — убедимся, что стоит "Без НДС"
				СтавкаБезНДС = VatRateBezNdsRef();
				ТЧСч = TryGetTablePart(СчетОбъект, "Запасы");
				Если (ТЧСч <> Неопределено) И (СтавкаБезНДС <> Неопределено) Тогда
					Если (ТЧСч.Количество() = 0) И (PreferredSum > 0) Тогда
						// если строк нет — попробуем создать одну строку из заказа
						ЗаказОбъект2 = Неопределено;
						Попытка ЗаказОбъект2 = ЗаказСсылка.ПолучитьОбъект(); Исключение ЗаказОбъект2 = Неопределено; КонецПопытки;
						ТЧЗ2 = TryGetTablePart(ЗаказОбъект2, "Запасы");
						Если (ТЧЗ2 <> Неопределено) И (ТЧЗ2.Количество() > 0) Тогда
							СтрЗ0 = Неопределено;
							Для Каждого S0 Из ТЧЗ2 Цикл СтрЗ0 = S0; Прервать; КонецЦикла;
							Если СтрЗ0 <> Неопределено Тогда
								СтрН = Неопределено;
								Попытка СтрН = ТЧСч.Добавить(); Исключение СтрН = Неопределено; КонецПопытки;
								Если СтрН <> Неопределено Тогда
									Попытка СтрН.Номенклатура = СтрЗ0.Номенклатура; Исключение КонецПопытки;
									Попытка СтрН.ЕдиницаИзмерения = СтрЗ0.ЕдиницаИзмерения; Исключение КонецПопытки;
									Попытка СтрН.Количество = 1; Исключение КонецПопытки;
									Попытка СтрН.Цена = PreferredSum; Исключение КонецПопытки;
									Попытка СтрН.Сумма = PreferredSum; Исключение КонецПопытки;
									Попытка СтрН.СтавкаНДС = СтавкаБезНДС; Исключение КонецПопытки;
									Попытка СтрН.СуммаНДС = 0; Исключение КонецПопытки;
									НужноПерепровести = Истина;
								КонецЕсли;
							КонецЕсли;
						КонецЕсли;
					КонецЕсли;

					Для Каждого СтрСч Из ТЧСч Цикл
						ТекСтавка = Неопределено;
						Попытка ТекСтавка = СтрСч.СтавкаНДС; Исключение ТекСтавка = Неопределено; КонецПопытки;
						Если НЕ ЗначениеЗаполнено(ТекСтавка) Тогда
							Попытка СтрСч.СтавкаНДС = СтавкаБезНДС; Исключение КонецПопытки;
							НужноПерепровести = Истина;
						КонецЕсли;
						Попытка СтрСч.СуммаНДС = 0; Исключение КонецПопытки;
						// Если по строке цена/сумма 0, а PreferredSum задан — поднимем сумму (иначе долг=0 и будет "мы должны")
						Если PreferredSum > 0 Тогда
							Кол0 = 0; Цена0 = 0; Сум0 = 0;
							Попытка Кол0 = СтрСч.Количество; Исключение Кол0 = 0; КонецПопытки;
							Попытка Цена0 = СтрСч.Цена; Исключение Цена0 = 0; КонецПопытки;
							Попытка Сум0 = СтрСч.Сумма; Исключение Сум0 = 0; КонецПопытки;
							Если (Сум0 = 0) И (Цена0 = 0) Тогда
								Если Кол0 = 0 Тогда Попытка СтрСч.Количество = 1; Исключение КонецПопытки; КонецЕсли;
								Попытка СтрСч.Цена = PreferredSum; Исключение КонецПопытки;
								Попытка СтрСч.Сумма = PreferredSum; Исключение КонецПопытки;
								Попытка СчетОбъект.СуммаДокумента = PreferredSum; Исключение КонецПопытки;
								НужноПерепровести = Истина;
							КонецЕсли;
						КонецЕсли;
					КонецЦикла;
				КонецЕсли;

				// Если что-то меняли — перепроведем даже если счет уже был проведен
				Если НужноПерепровести Тогда
					Попытка
						СчетОбъект.Записать(РежимЗаписиДокумента.Проведение);
						Счет = СчетОбъект.Ссылка;
					Исключение
						Счет = Неопределено;
					КонецПопытки;
				КонецЕсли;

				Проведен = Ложь;
				Попытка Проведен = СчетОбъект.Проведен; Исключение Проведен = Ложь; КонецПопытки;
				Если НЕ Проведен Тогда
					Попытка
						СчетОбъект.Записать(РежимЗаписиДокумента.Проведение);
						Счет = СчетОбъект.Ссылка;
					Исключение
						// если не провёлся — будем создавать новый ниже
						Счет = Неопределено;
					КонецПопытки;
				Иначе
					// если проведен, но сумма осталась 0 — считаем счет непригодным (долга нет => платеж выглядит как переплата)
					СуммаСчета2 = 0;
					Попытка СуммаСчета2 = СчетОбъект.СуммаДокумента; Исключение СуммаСчета2 = 0; КонецПопытки;
					Если СуммаСчета2 = 0 Тогда
						Счет = Неопределено;
					КонецЕсли;
				КонецЕсли;
				// после проведения тоже проверим сумму
				Если ЗначениеЗаполнено(Счет) Тогда
					СуммаСчета3 = 0;
					Попытка СуммаСчета3 = СчетОбъект.СуммаДокумента; Исключение СуммаСчета3 = 0; КонецПопытки;
					Если СуммаСчета3 = 0 Тогда
						Счет = Неопределено;
					КонецЕсли;
				КонецЕсли;
				Если ЗначениеЗаполнено(Счет) Тогда
					Р.ok = Истина;
					Р.posted = Истина;
					Р.ref = Счет;
					Возврат Р;
				КонецЕсли;
			Иначе
				Счет = Неопределено;
			КонецЕсли;
		Иначе
			Счет = Неопределено;
		КонецЕсли;
	КонецЕсли;

	// 2) Создаем новый счет на оплату
	ЗаказОбъект = Неопределено;
	Попытка ЗаказОбъект = ЗаказСсылка.ПолучитьОбъект(); Исключение ЗаказОбъект = Неопределено; КонецПопытки;
	Если ЗаказОбъект = Неопределено Тогда Р.error = "no_order_object"; Возврат Р; КонецЕсли;

	Док = Неопределено;
	Попытка Док = Документы.СчетНаОплату.СоздатьДокумент(); Исключение Док = Неопределено; КонецПопытки;
	Если Док = Неопределено Тогда Р.error = "no_doc_СчетНаОплату"; Возврат Р; КонецЕсли;

	Попытка Док.Дата = ТекущаяДата(); Исключение КонецПопытки;
	Попытка Док.ДокументОснование = ЗаказСсылка; Исключение КонецПопытки;
	Попытка Док.Организация = ЗаказОбъект.Организация; Исключение КонецПопытки;
	Попытка Док.Контрагент = ЗаказОбъект.Контрагент; Исключение КонецПопытки;
	Попытка Док.Договор = ЗаказОбъект.Договор; Исключение КонецПопытки;

	// Валюта/курс/кратность
	Попытка Док.ВалютаДокумента = ЗаказОбъект.ВалютаДокумента; Исключение КонецПопытки;
	Попытка Док.Курс = ЗаказОбъект.Курс; Исключение КонецПопытки;
	Попытка Док.Кратность = ЗаказОбъект.Кратность; Исключение КонецПопытки;
	К0 = 1; Кр0 = 1;
	Попытка К0 = Док.Курс; Исключение К0 = 1; КонецПопытки;
	Попытка Кр0 = Док.Кратность; Исключение Кр0 = 1; КонецПопытки;
	Если К0 = 0 Тогда К0 = 1; КонецЕсли;
	Если Кр0 = 0 Тогда Кр0 = 1; КонецЕсли;
	Попытка Док.Курс = К0; Исключение КонецПопытки;
	Попытка Док.Кратность = Кр0; Исключение КонецПопытки;

	// Сумма: если в заказе 0/пусто — используем PreferredSum (иначе долг не формируется)
	СуммаДок = 0;
	Попытка СуммаДок = ЗаказОбъект.СуммаДокумента; Исключение СуммаДок = 0; КонецПопытки;
	Если СуммаДок = 0 Тогда Попытка СуммаДок = ЗаказОбъект.Сумма; Исключение СуммаДок = 0; КонецПопытки; КонецЕсли;
	Если (СуммаДок = 0) И (PreferredSum > 0) Тогда СуммаДок = PreferredSum; КонецЕсли;
	Попытка Док.СуммаДокумента = СуммаДок; Исключение КонецПопытки;

	// Запасы: копируем строки из заказа, чтобы счет мог проведется и имел корректную сумму
	ТЧЗаказ = TryGetTablePart(ЗаказОбъект, "Запасы");
	ТЧСчет = TryGetTablePart(Док, "Запасы");
	Если (ТЧЗаказ <> Неопределено) И (ТЧСчет <> Неопределено) Тогда
		Попытка ТЧСчет.Очистить(); Исключение КонецПопытки;
		// Ставка НДС обязательна в строках счета (FillChecking=ShowError) — подберем "Без НДС"
		СтавкаБезНДС = VatRateBezNdsRef();

		СуммаПоСтрокам = 0;
		ПерваяСтрокаУсилена = Ложь;
		Для Каждого СтрЗ Из ТЧЗаказ Цикл
			СтрС = Неопределено;
			Попытка СтрС = ТЧСчет.Добавить(); Исключение СтрС = Неопределено; КонецПопытки;
			Если СтрС = Неопределено Тогда Продолжить; КонецЕсли;
			Попытка СтрС.Номенклатура = СтрЗ.Номенклатура; Исключение КонецПопытки;
			Кол = 0;
			Попытка Кол = СтрЗ.Количество; Исключение Кол = 0; КонецПопытки;
			Попытка СтрС.Количество = Кол; Исключение КонецПопытки;
			Попытка СтрС.ЕдиницаИзмерения = СтрЗ.ЕдиницаИзмерения; Исключение КонецПопытки;
			Цена = 0;
			Попытка Цена = СтрЗ.Цена; Исключение Цена = 0; КонецПопытки;
			Попытка СтрС.Цена = Цена; Исключение КонецПопытки;
			Сум = 0;
			Попытка Сум = СтрЗ.Сумма; Исключение Сум = 0; КонецПопытки;
			Если Сум = 0 Тогда Сум = Кол * Цена; КонецЕсли;
			// Если по заказу суммы/цены нулевые, то для формирования долга по счету поднимем первую строку до PreferredSum
			Если (Сум = 0) И (PreferredSum > 0) И (НЕ ПерваяСтрокаУсилена) Тогда
				ПерваяСтрокаУсилена = Истина;
				Кол = 1;
				Цена = PreferredSum;
				Сум = PreferredSum;
				Попытка СтрС.Количество = Кол; Исключение КонецПопытки;
				Попытка СтрС.Цена = Цена; Исключение КонецПопытки;
			КонецЕсли;
			Попытка СтрС.Сумма = Сум; Исключение КонецПопытки;
			СуммаПоСтрокам = СуммаПоСтрокам + Сум;
			Если СтавкаБезНДС <> Неопределено Тогда
				Попытка СтрС.СтавкаНДС = СтавкаБезНДС; Исключение КонецПопытки;
			КонецЕсли;
			Попытка СтрС.СуммаНДС = 0; Исключение КонецПопытки;
		КонецЦикла;
		Если (СуммаДок = 0) И (СуммаПоСтрокам > 0) Тогда
			СуммаДок = СуммаПоСтрокам;
			Попытка Док.СуммаДокумента = СуммаДок; Исключение КонецПопытки;
		КонецЕсли;
	КонецЕсли;

	Попытка
		Док.Записать(РежимЗаписиДокумента.Проведение);
		Р.ok = Истина;
		Р.posted = Истина;
		Р.ref = Док.Ссылка;
	Исключение
		Р.error = ОписаниеОшибки();
	КонецПопытки;

	Возврат Р;
КонецФункции
